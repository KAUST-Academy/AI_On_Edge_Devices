import numpy as np

def window(samples, size=100):
    """Cut the signal into windows of a fixed length."""
    for start in range(0, len(samples) - size, size):
        yield np.array(samples[start:start + size])

% ------------------------------------------------------ exercise and answer ---
\exerciseframe{model size}{%
  A model uses 180 KB of flash and 90 KB of RAM.\\
  The board has 8 MB of flash and 512 KB of SRAM.
  \begin{enumerate}
    \item Does the model fit?
    \item Which number limits the deployment?
  \end{enumerate}
}

\answerframe{model size}{%
  \begin{enumerate}
    \item Yes. Flash and RAM are both large enough.
    \item RAM. The arena, the stack, and the buffers of the drivers share the
          512 KB. Flash holds only the program and the model.
  \end{enumerate}
}

\begin{frame}[fragile]{Exercise block with a listing}
  \begin{exerciseblock}
    Read the line below. Which change makes the buffer four times smaller?
    \begin{lstlisting}[style=cpp]
int16_t buffer[200];
    \end{lstlisting}
  \end{exerciseblock}
  \vskip 0.4em
  \begin{answerblock}
    Store the samples as \code{int8\_t} and halve the window length.
  \end{answerblock}
\end{frame}
