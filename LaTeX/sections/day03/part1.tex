\begin{frame}{What a model file holds}
  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      \begin{block}{In the file}
        \begin{itemize}
          \item The graph: the operators and their order.
          \item The tensors: shape and data type of each input, output,
                and weight.
          \item The weights: the trained values.
          \item The parameters of the quantization (Day 4).
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{block}{Not in the file}
        \begin{itemize}
          \item The Python code of the model.
          \item The optimizer and the loss function.
          \item The gradients.
          \item Layers that work only in training, for example dropout.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
  \vskip 0.6em
  \begin{itemize}
    \item A deployment format describes only the forward pass.
    \item The runtime needs no knowledge of the framework that trained the
          model.
  \end{itemize}
\end{frame}

    \draw[->, cardinalred] (5.4,1.0) to[bend left=50] (8.0,1.0);
    \node[cardinalred] at (5.0,2.35) {offsets: the position of the next
      part, in bytes};
    \draw[<->, coolgray] (0,-0.3) -- node[below, text=black]
      {one array of bytes in the flash} (10,-0.3);
  \end{tikzpicture}\par\endgroup
  \vskip 0.3em
  \small
  \begin{itemize}
    \item FlatBuffers is a storage format. Each part holds the offset of
          the parts that it uses.
    \item The runtime reads a value at its position in the array. It makes
          no copy and builds no objects.
    \item So the weights stay in the flash. The RAM holds only the
          activations.
  \end{itemize}
  \keypoint{A format that needs a parsing step makes a second copy of the
    model in RAM. A microcontroller has no RAM for that copy.}
\end{frame}

    needs. A small model uses three or four operator types.}

% --------------------------------------------------------------------- Keras
\begin{frame}[fragile]{Keras to LiteRT}
\begin{lstlisting}[style=python]
import keras
import tensorflow as tf

model = keras.Sequential([
    keras.Input(shape=(300,)),               # 2 s at 50 Hz, 3 axes
    keras.layers.Dense(20, activation="relu"),
    keras.layers.Dense(10, activation="relu"),
    keras.layers.Dense(4, activation="softmax"),
])
# ... train the model ...

converter = tf.lite.TFLiteConverter.from_keras_model(model)
tflite_model = converter.convert()           # bytes of the LiteRT file
with open("model.tflite", "wb") as handle:
    handle.write(tflite_model)
\end{lstlisting}
  \small
  \begin{itemize}
    \item The model has 6274 parameters. With \code{float32}, the weights
          need $6274 \times 4 = 25\,096$ bytes.
    \item The LiteRT file has 26\,732 bytes: the weights, and 1636 bytes
          for the graph and the tensor descriptions.
  \end{itemize}
  \source{File size: result of the converter of TensorFlow 2.21.0 for this
    model.}
\end{frame}

\begin{frame}{What the converter does}
  \begingroup\centering
  \begin{tikzpicture}[font=\footnotesize, >={Stealth[length=5pt]},
      k/.style={draw=coolgray, rounded corners=2pt, fill=boxgray,
                minimum width=1.5cm, minimum height=0.6cm, align=center},
      t/.style={k, draw=cardinalred, minimum width=3.5cm}]
    \node[anchor=west, text=coolgray] at (-0.6,0.75) {Keras: 6 steps};
    \node[anchor=west, text=coolgray] at (-0.6,-0.9)
      {LiteRT: 4 operators (FC = FULLY\_CONNECTED)};
    \node[k, minimum width=1.3cm] (a1) at (0.05,0)   {Dense};
    \node[k, minimum width=1.3cm] (a2) at (1.55,0) {ReLU};
    \node[k, minimum width=1.3cm] (a3) at (3.35,0) {Dense};
    \node[k, minimum width=1.3cm] (a4) at (4.85,0) {ReLU};
    \node[k, minimum width=1.3cm] (a5) at (6.65,0) {Dense};
    \node[k, minimum width=1.3cm] (a6) at (8.45,0) {Softmax};
    \node[t, minimum width=2.8cm] (b1) at (0.8,-1.7) {FC with ReLU};
    \node[t, minimum width=2.8cm] (b2) at (4.1,-1.7) {FC with ReLU};
    \node[t, minimum width=1.3cm] (b3) at (6.65,-1.7) {FC};
    \node[t, minimum width=1.3cm] (b4) at (8.45,-1.7) {SOFTMAX};
    \foreach \a/\b in {a1/a2, a2/a3, a3/a4, a4/a5, a5/a6} {
      \draw[->] (\a) -- (\b); }
    \draw[->, coolgray] (0.8,-0.35) -- (0.8,-0.6);
    \draw[->, coolgray] (4.1,-0.35) -- (4.1,-0.6);
    \draw[->, coolgray] (6.65,-0.35) -- (6.65,-0.6);
    \draw[->, coolgray] (8.45,-0.35) -- (8.45,-0.6);
  \end{tikzpicture}\par\endgroup
  \vskip 0.5em
  \begin{itemize}
    \item \textbf{Freeze:} each variable becomes a constant.
    \item \textbf{Remove:} the converter removes the operations that only
          the training uses.
    \item \textbf{Fuse:} an activation function becomes a part of the
          operator before it. A batch normalization becomes a part of the
          weights of the convolution before it.
    \item \textbf{Map:} each layer becomes one operator of the LiteRT list.
  \end{itemize}
  \keypoint{The example has six Keras steps and four LiteRT operators. Fewer
    operators give fewer tensors in RAM.}
\end{frame}

\begin{frame}[fragile]{Check the converted model before you deploy it}
\begin{lstlisting}[style=python]
import numpy as np
from ai_edge_litert.interpreter import Interpreter

interpreter = Interpreter(model_path="model.tflite")
interpreter.allocate_tensors()
inp = interpreter.get_input_details()[0]

\end{lstlisting}
  \small
  \begin{itemize}
    \item The LiteRT interpreter runs on the laptop, with the same file
          that goes to the device.
    \item Compare the two outputs for the same input. Then compare the
          accuracy on the test set. Do this after each conversion step.
  \end{itemize}
  \source{Difference: result for the example model with TensorFlow 2.21.0.}
\end{frame}

\begin{frame}[fragile]{From the file to a C array}
\begin{lstlisting}[style=shell]
python3 tflite_to_header.py model.tflite model.h
\end{lstlisting}
\begin{lstlisting}[style=cpp]
// Model: model.tflite (26732 bytes)
alignas(16) const unsigned char g_model[] = {
    0x1c, 0x00, 0x00, 0x00, 0x54, 0x46, 0x4c, 0x33, 0x14, 0x00, 0x20, 0x00,
    ...
};
const int g_model_len = sizeof(g_model);
\end{lstlisting}
  \small
  \begin{itemize}
    \item The sketch includes \code{model.h}. The model is then a part of
          the program.
    \item \code{const}: the array stays in the flash. With no \code{const},
          the program copies it to RAM at the start.
    \item \code{alignas(16)}: the runtime reads numbers of 4 and 8 bytes at
          fixed positions. The array must start at a correct address.
    \item The bytes 4 to 7 are \code{TFL3}: the identifier of a LiteRT
          file.
  \end{itemize}
  \source{The script is in \texttt{Labs/hardware/HW-02/}. The command
    \texttt{xxd -i} gives the same bytes.}
\end{frame}

\begin{frame}{Conversion errors and their causes}
  \small
  \renewcommand{\arraystretch}{1.3}
  \begingroup\centering
  \begin{tabular}{@{}L{0.25\textwidth}L{0.31\textwidth}L{0.34\textwidth}@{}}
    \toprule
    \textbf{Cause} & \textbf{Example} & \textbf{Correction} \\
    \midrule
    An operator with no kernel & A new activation function, a custom layer
        & Use a layer from the supported list \\
    A shape that is not fixed & An input of variable length
        & Fix the input shape before the conversion \\
    Control flow & A loop or a condition that depends on the data
        & Write the model with no such step \\
    A data type with no kernel & \code{float64}, or \code{int8} for an
          operator that has only a \code{float32} kernel
        & Change the type, or change the operator \\
    \bottomrule
  \end{tabular}\par\endgroup
  \vskip 0.5em
  \keypoint{Read the operator list of the target runtime before you select
    the layers of the model.}
\end{frame}

\begin{frame}[fragile]{An error can come on the device}
  \small
  The conversion gives no error. The sketch compiles. On the board, the
  serial port shows:
\begin{lstlisting}
Didn't find op for builtin opcode 'SOFTMAX' version '1'.
Failed to get registration from op code SOFTMAX
\end{lstlisting}
  \begin{itemize}
    \item The file is correct. The \emph{program} has no kernel for one
          operator of the model.
    \item With TensorFlow Lite Micro, the program names each operator that
          it needs. This list is the operator resolver. Part 2 explains it.
    \item An operator that the program does not name is not in the
          firmware. This keeps the firmware small.
  \end{itemize}
  \vskip 0.3em
  \begin{block}{Three places for an operator error}
    \begin{enumerate}
      \item At the conversion: the converter has no mapping.
      \item At the build: the library has no kernel.
      \item On the device: the resolver of the sketch does not name the
            operator.
    \end{enumerate}
  \end{block}

\answerframe{read a LiteRT file in Netron}{%
  \small
  \begin{enumerate}
    \item Three operators: FullyConnected, FullyConnected, FullyConnected.
          The first two have a fused ReLU activation.
    \item The input has the shape (1, 1) and the type \code{int8}. The
          output has the same shape and the same type.
    \item The weights have the type \code{int8}. The biases have the type
          \code{int32}. The model is quantized. Day 4 explains why the bias
          has more bits.
    \item Weights: $16 + 256 + 16 = 288$. Biases: $16 + 16 + 1 = 33$. The
          model has 321 parameters. They need
          $288 \times 1 + 33 \times 4 = 420$ bytes.
    \item $420 / 2488 = 17$ percent. For a very small model, the
          description of the graph is larger than the weights. For the
          example model of this part, the weights are 94 percent of the
          file.
  \end{enumerate}
}
