% Motivation and learning outcomes.
% The outcomes are the outcomes of Day 15 in the syllabus.
\begin{frame}{Why this day matters}
  \begin{itemize}
    \item This is the last day of the course. Part 1 of the lecture reviews
          the course in one view: the trade-offs that each day measured.
    \item After Part 1, the lecture time is for the capstone build. Your team
          builds, and the instructor visits each team.
    \item In the lab, each team completes the measurements, gives a
          demonstration of 10 minutes, and submits the report.
    \item A good demonstration shows a working system and defends each
          decision with a measured number.
  \end{itemize}
\end{frame}

\begin{frame}{Learning outcomes}
  After this day you can:
  \begin{itemize}
    \item Integrate models, devices, messages, and monitoring into one system.
    \item Report system performance with measured numbers.
    \item Present and defend design decisions.
  \end{itemize}
\end{frame}
