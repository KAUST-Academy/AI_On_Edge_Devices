% ---------------------------------------------------------------------- IoU
\begin{frame}{IoU: the measure for the place of a box}
  \begingroup\centering
  \begin{tikzpicture}[font=\footnotesize,
      lab/.style={treegreen, dashed, thick},
      pre/.style={cardinalred, thick}]
    \foreach \k/\s/\v in {0/0.068/0.9, 1/0.229/0.7, 2/0.433/0.5,
                          3/0.700/0.3} {
      \fill[treegreen!12] (\k*2.6,0) rectangle (\k*2.6+1.3,1.3);
      \draw[lab] (\k*2.6,0) rectangle (\k*2.6+1.3,1.3);
      \draw[pre] (\k*2.6+\s,0) rectangle (\k*2.6+1.3+\s,1.3);
      \node[below] at (\k*2.6+0.65+0.5*\s,0) {IoU \v};
    }
  \end{tikzpicture}\par\endgroup
  \vskip 0.4em
  \small
  \begin{itemize}
    \item Part 1 used the IoU for two candidates. The metrics use it for a
          prediction and a label.
    \item A prediction is in the correct place when its IoU with the label
          is at or above a threshold.
    \item The usual threshold is 0.5. In the drawing, a box with the IoU
          0.5 does not look good: one third of the label is not in the
          box.
    \item A second, more strict rule uses ten thresholds from 0.5 to 0.95.
  \end{itemize}
  \keypoint{The IoU threshold of a metric is not the IoU threshold of the
    suppression. Name the threshold with each result.}
\end{frame}

\begin{frame}{A small object needs an exact box}
  \begin{columns}[c]
    \begin{column}{0.46\textwidth}
      \small
      \renewcommand{\arraystretch}{1.2}
      \begin{tabular}{@{}rrr@{}}
        \toprule
        \textbf{Side of the box} & \textbf{Error} & \textbf{IoU} \\
        \midrule
        100 pixels & 4 pixels & 0.92 \\
        40 pixels  & 4 pixels & 0.82 \\
        20 pixels  & 4 pixels & 0.67 \\
        10 pixels  & 4 pixels & 0.43 \\
        \bottomrule
      \end{tabular}
      \[
        \text{IoU} = \frac{L - s}{L + s}
      \]
    \end{column}
    \begin{column}{0.52\textwidth}
      \small
      \begin{itemize}
        \item A square box with the side $L$ moves by $s$ pixels to the
              right.
        \item The same error of 4 pixels is small for a large object. For
              an object of 10 pixels, the prediction is not correct at the
              threshold 0.5.
        \item The same rule applies to the label. A label that is 4 pixels
              wrong makes a correct prediction look wrong.
      \end{itemize}
    \end{column}
  \end{columns}
  \vskip 0.4em
  \keypoint{The metrics are strict for small objects. A larger input
    resolution gives each object more pixels (Part 3).}
\end{frame}

% ------------------------------------------------------------------ matching
\begin{frame}{True positive, false positive, false negative}
  \begin{columns}[c]
    \begin{column}{0.40\textwidth}
      \centering
      \begin{tikzpicture}[font=\scriptsize, x=0.42cm, y=0.42cm,
          lab/.style={treegreen, dashed, thick},
          pre/.style={cardinalred, thick}]
        \fill[boxgray] (0,0) rectangle (10,8);
        \draw[lab] (0.7,4.0) rectangle (3.3,7.2);
        \draw[pre] (0.8,3.9) rectangle (3.4,7.1);
        \draw[pre] (1.5,3.4) rectangle (4.2,6.6);
        \draw[lab] (5.8,4.6) rectangle (9.3,7.2);
        \draw[pre] (6.0,4.5) rectangle (9.4,7.0);
        \draw[lab] (0.7,0.4) rectangle (3.1,2.4);
        \draw[pre] (6.2,0.5) rectangle (8.8,2.5);
        \node[cardinalred] at (2.0,7.6) {0.9: TP};
        \node[cardinalred, anchor=west] at (4.25,3.7) {0.8: FP};
        \node[cardinalred] at (7.6,7.6) {0.7: TP};
        \node[cardinalred] at (7.5,1.5) {0.6: FP};
        \node[treegreen!60!black] at (1.9,1.4) {FN};
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.58\textwidth}
      \small
      \begin{enumerate}
        \item Sort the predictions by their score.
        \item Take the next prediction. Find the label of the same class
              with the largest IoU that is still free.
        \item If this IoU is at or above the threshold, the prediction is
              a \textbf{true positive} (TP). The label is not free now.
        \item If not, the prediction is a \textbf{false positive} (FP).
        \item A label with no prediction is a \textbf{false negative}
              (FN).
      \end{enumerate}
    \end{column}
  \end{columns}
  \vskip 0.3em
  \small
  \begin{itemize}
    \item A second box for one object is a false positive. A detector with
          no suppression gets a bad result.
    \item A detector has no true negatives: the number of places with no
          object has no limit. So nobody reports the accuracy of a detector.
  \end{itemize}
\end{frame}

% ------------------------------------------------------- precision, recall
\begin{frame}{Precision and recall}
  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \begin{block}{Precision}
        \[
          \frac{TP}{TP + FP} = \frac{713}{713 + 213} = 0.77
        \]
        \small
        Of the boxes that the model gives, the part that is correct.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Recall}
        \[
          \frac{TP}{TP + FN} = \frac{713}{713 + 356} = 0.67
        \]
        \small
        Of the objects in the images, the part that the model finds.
      \end{block}
    \end{column}
  \end{columns}
  \vskip 0.5em
  \small
  \begin{itemize}
    \item The example: YOLO11n and the class ``person'' in 500 images, with
          the score threshold 0.25 and the IoU threshold 0.5. The images
          have 1069 labels of this class.
    \item The model gives 926 boxes. 713 are correct, and 213 are false
          alarms. It does not find 356 persons.
    \item Low precision: the product reports objects that are not there.
          Low recall: the product does not see objects.
  \end{itemize}
  \source{An experiment of this course with 500 images of COCO.}
\end{frame}

\begin{frame}{The score threshold: precision against recall}
  \small
  \renewcommand{\arraystretch}{1.25}
  \begin{tabular}{@{}rrrrrr@{}}
    \toprule
    \textbf{Score threshold} & \textbf{TP} & \textbf{FP} & \textbf{FN}
    & \textbf{Precision} & \textbf{Recall} \\
    \midrule
    0.10 & 819 & 677 & 250 & 0.55 & 0.77 \\
    0.25 & 713 & 213 & 356 & 0.77 & 0.67 \\
    0.50 & 560 & 45  & 509 & 0.93 & 0.52 \\
    0.75 & 337 & 13  & 732 & 0.96 & 0.32 \\
    \bottomrule
  \end{tabular}
  \vskip 0.5em
  \begin{itemize}
    \item The same model, the same images, the class ``person''. Only the
          score threshold changes.
    \item A higher threshold removes false positives: the precision goes
          up. It also removes true positives: the recall goes down.
    \item With the threshold 0.001, the model gives 13\,032 boxes for 1069
          persons. The recall is 0.90: no threshold finds each person.
  \end{itemize}
  \keypoint{One value of the precision or of the recall says nothing with
    no score threshold. Report the two values and the threshold together.}
  \source{An experiment of this course with 500 images of COCO.}
\end{frame}

  \end{columns}
  \small
  \begin{itemize}
    \item One text file for each image, with the same name. One line for
          each object: the class, then $c_x, c_y, w, h$ as parts of the
          image from 0 to 1.
    \item An image with no object has an empty file or no file. Such
          images teach the model the background.
    \item \code{data.yaml} names the folders and the classes. The class 0
          is the first name.
    \item Three sets: train, validation for the decisions during the work,
          test for the final result.
  \end{itemize}
  \source{Format: kit lab ``Object Detection''.}
\end{frame}

\begin{frame}{Measured: label errors change the result}
  \footnotesize
  \renewcommand{\arraystretch}{1.2}
  \begin{tabular}{@{}L{0.42\textwidth}rrrr@{}}
    \toprule
    \textbf{Labels of the 500 test images} & \textbf{Precision}
    & \textbf{Recall} & \textbf{mAP50} & \textbf{mAP50-95} \\
    \midrule
    The labels of COCO                 & 0.674 & 0.530 & 0.574 & 0.417 \\
    Sides moved by up to 5 percent     & 0.669 & 0.528 & 0.570 & 0.379 \\
    Sides moved by up to 10 percent    & 0.666 & 0.522 & 0.562 & 0.315 \\
    Sides moved by up to 20 percent    & 0.645 & 0.499 & 0.530 & 0.184 \\
    10 percent of the labels removed   & 0.609 & 0.525 & 0.530 & 0.386 \\
    30 percent of the labels removed   & 0.473 & 0.516 & 0.434 & 0.312 \\
    10 percent with a wrong class      & 0.618 & 0.414 & 0.425 & 0.310 \\
    \bottomrule
  \end{tabular}
  \vskip 0.4em
  \small
  \begin{itemize}
    \item The model and the images are the same in each row. Only the
          labels change. A side moves by a random part of the size of its
          box.
    \item Loose boxes change mAP50 little. They cut mAP50-95 by more than
          one half.
    \item Missing labels cut the precision: the model finds objects that
          have no label.
  \end{itemize}
  \keypoint{With bad test labels, you cannot measure a good model.}
  \source{An experiment of this course with YOLO11n.}
\end{frame}

\begin{frame}{How to check the labels}
  \small
  \renewcommand{\arraystretch}{1.25}
  \begingroup\centering
  \begin{tabular}{@{}L{0.42\textwidth}L{0.50\textwidth}@{}}
    \toprule
    \textbf{Check} & \textbf{It finds} \\
    \midrule
    Draw the labels on the images. Look at a sample of each class.
        & Loose boxes, wrong classes \\
    Count the labels of each class, and the images with no label.
        & A class with too few examples, a typing error in a class name \\
    Plot the widths and the heights of all boxes.
        & Boxes with a size of zero, boxes as large as the image \\
    Train a first model. Look at its false positives with the highest
    score.
        & Missing labels: the model is correct, and the label is not
          there \\

    boxes.
        & Rules that are not clear \\
    \bottomrule
  \end{tabular}\par\endgroup

    code. After a correction, train again and measure again.}

      \begin{tabular}{@{}lll@{}}
        \toprule
        & \textbf{Label} & \textbf{Prediction} \\
        \midrule

    \item Calculate the IoU of each pair.
    \item Is each prediction a true positive with the IoU threshold 0.5?

  \end{enumerate}

\answerframe{the IoU of two box pairs}{%
  \small
  \renewcommand{\arraystretch}{1.2}
  \begin{tabular}{@{}lrrrll@{}}
    \toprule
    & \textbf{Intersection} & \textbf{Union} & \textbf{IoU}
    & \textbf{At 0.5} & \textbf{At 0.75} \\
    \midrule
    Pair 1 & $8 \times 10 = 80$ & $100 + 100 - 80 = 120$ & 0.67
           & TP & FP \\
    Pair 2 & $2 \times 4 = 8$   & $16 + 16 - 8 = 24$     & 0.33
           & FP & FP \\
    \bottomrule
  \end{tabular}
  \vskip 0.4em
  \begin{enumerate}
    \item The IoU is 0.67 and 0.33.
    \item Pair 1 is a true positive at 0.5 only. Pair 2 is a false
          positive at the two thresholds. The label of pair 2 is then a
          false negative also: one error of the place gives two errors of
          the count.
    \item The error is 20 percent of the side of the first box, and 50
          percent of the side of the second box. The IoU measures the
          error as a part of the size of the object.
    \item No. Its IoU is $90 / 110 = 0.82$, which is better than the first
          box. But the first box has the higher score and takes the label.
          The second box is a false positive. The suppression of Part 1
          must remove it.
  \end{enumerate}
}
