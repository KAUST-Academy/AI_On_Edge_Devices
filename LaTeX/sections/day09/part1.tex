% ---------------------------------------------------------------- throughput
\begin{frame}{Throughput is not always 1 / latency}
  \small
  \renewcommand{\arraystretch}{1.25}
  \begin{tabular}{@{}L{0.44\textwidth}rr@{}}
    \toprule
    \textbf{MobileNetV2 on 4 cores} & \textbf{Median latency}
    & \textbf{Throughput} \\
    \midrule
    One stream, 1 thread             & 11.49 ms & 86.4 each second \\
    One stream, 4 threads            & 3.26 ms  & 300.8 each second \\
    Four streams, 1 thread each      & 12.43 ms & 321.9 each second \\
    \bottomrule
  \end{tabular}
  \vskip 0.5em
  \begin{itemize}
    \item \textbf{Latency}: the time for one answer. \textbf{Throughput}:
          the answers in one second, for the complete device.
    \item One stream: the throughput is about 1 / latency. Four threads make
          one inference 3.5 times as fast, not 4 times. Each row is a run
          of 10 seconds.
    \item Four streams at the same time give 7 percent more throughput. But
          each inference needs 3.8 times as long.
    \item One camera with a person in front of it needs a low latency. A
          recorder with four cameras needs a high throughput.
  \end{itemize}
  \keypoint{Decide which of the two numbers your product needs. Then set
    the threads and the streams for that number.}
  \source{An experiment of this course on an x86 processor.}
\end{frame}

% -------------------------------------------------------------------- memory
\begin{frame}{Memory: three different numbers}
  \small
  \renewcommand{\arraystretch}{1.25}
  \begin{tabular}{@{}L{0.50\textwidth}rr@{}}
    \toprule
    \textbf{MobileNetV2 in LiteRT} & \textbf{\code{float32}}
    & \textbf{\code{int8}} \\
    \midrule
    Model file (storage)                      & 13.4 MB & 3.9 MB \\
    RAM that the load of the model adds       & 47.3 MB & 24.6 MB \\
    Peak RAM that the model adds              & 48.2 MB & 25.0 MB \\
    \bottomrule
  \end{tabular}
  \vskip 0.5em
  \begin{itemize}
    \item The Python process with the runtime has 31 MB before the load.
    \item The runtime makes its own copy of the weights and adds the buffers
          of the activations. The RAM is 3.5 times the \code{float32} file.
    \item Linux: read the peak of the process (\code{VmHWM} in
          \code{/proc/self/status}). A computer with 8 GB has room.
    \item Microcontroller: the flash holds the program and the model. The
          RAM holds the arena and the other data. No room: no run.
  \end{itemize}
  \keypoint{The file size does not tell you the RAM. Measure the peak RAM
    of the complete program on the device.}
  \source{An experiment of this course. 1 MB is 1024 KB.}
\end{frame}

% -------------------------------------------------------------------- energy
\begin{frame}{Energy per inference}
  \begin{columns}[T]
    \begin{column}{0.40\textwidth}
      \begin{block}{Definition}
        \small
        $E = P_{\text{active}} \times t_{\text{inference}}$\\[3pt]
        1 mW $\times$ 1 ms = 1 $\mu$J
      \end{block}
    \end{column}
    \begin{column}{0.56\textwidth}
      \begin{block}{A published result}
        \small
        Keyword spotting on a Cortex-M4 board:\\
        43.19 ms and 1820.5 $\mu$J for one inference.\\
        Mean power: $1820.5 / 43.19 = 42.2$ mW.
      \end{block}
    \end{column}
  \end{columns}
  \vskip 0.5em
  \small
  \begin{itemize}
    \item The power (mW) says how fast the device uses its battery at this
          moment. The energy ($\mu$J) says what one inference costs.
    \item A battery holds energy. The number of inferences of one battery
          is its energy divided by the energy per inference.
    \item A faster processor can use more power and less energy: it
          completes the work earlier. Speed and efficiency are two
          different results.
    \item A device that sleeps between two inferences also uses energy in
          the sleep: Part 3.
  \end{itemize}
  \source{MLPerf Tiny v1.2, board NUCLEO-L4R5ZI (MLCommons).\\
    The latency and the power are calculations.}
\end{frame}

\begin{frame}{How many test samples?}
  \begin{columns}[c]
    \begin{column}{0.36\textwidth}
      \small
      \renewcommand{\arraystretch}{1.15}
      \begin{tabular}{@{}rr@{}}
        \toprule
        \textbf{Samples} & \textbf{Interval} \\
        \midrule
        100     & $\pm$ 8.2 points \\
        200     & $\pm$ 5.8 points \\
        500     & $\pm$ 3.7 points \\
        1000    & $\pm$ 2.6 points \\
        5000    & $\pm$ 1.2 points \\
        50\,000 & $\pm$ 0.4 points \\
        \bottomrule
      \end{tabular}
    \end{column}
    \begin{column}{0.60\textwidth}
      \small
      \[
        \pm\, 1.96 \sqrt{\frac{p\,(1 - p)}{N}}
      \]
      \begin{itemize}
        \item The 95 percent interval for an accuracy $p$ of 77.4 percent
              and $N$ test samples.
        \item 77.4 and 75.2 percent on 500 images: the difference of 2.2
              points is smaller than the interval of one value.
        \item For $\pm$ 1 point you need 6720 samples.
      \end{itemize}
    \end{column}
  \end{columns}
  \vskip 0.4em
  \small
  \begin{itemize}
    \item Two files on the same images: count the images for which the two
          answers differ. This comparison is more exact than two intervals.
    \item MLPerf Tiny tests the image classifier with 200 images. Its
          quality target (85 percent) is below the reference (86.5 percent).
  \end{itemize}
  \keypoint{A test set is a measuring instrument. Its size sets the
    smallest difference that you can see.}
  \source{A calculation with the normal approximation.\\
    MLPerf Tiny: Banbury et al., 2021.}
\end{frame}

\begin{frame}{Why a standard benchmark suite}
  \begin{columns}[T]
    \begin{column}{0.40\textwidth}
      \begin{block}{Five vendors, five claims}
        \footnotesize
        \begin{itemize}
          \item A: ``8 TOPS with \code{int8}''
          \item B: ``15 TOPS with 4 bits''
          \item C: the latency of its own model
          \item D: a result of an old benchmark
          \item E: the peak throughput
        \end{itemize}
        You cannot compare two of them.
      \end{block}
    \end{column}
    \begin{column}{0.57\textwidth}
      \footnotesize
      \renewcommand{\arraystretch}{1.15}
      \setlength{\tabcolsep}{3pt}
      \begin{tabular}{@{}L{1.5cm}L{2.2cm}L{2.0cm}@{}}
        \toprule
        \textbf{Suite} & \textbf{Devices} & \textbf{Main metrics} \\
        \midrule
        Training          & Data centre
            & Time to a quality target \\
        Inference, Datacenter & Servers
            & Throughput with a latency limit \\
        Inference, Edge   & Systems outside a data centre
            & Latency percentile, throughput \\
        Tiny              & Microcontrollers
            & Latency, accuracy, energy \\
        \bottomrule
      \end{tabular}
    \end{column}
  \end{columns}
  \vskip 0.4em
  \small
  \begin{itemize}
    \item A suite fixes the parts of the benchmark for all submitters: the
          task, the dataset, the model, the quality target, the harness, and
          the run rules.
    \item MLCommons, a non-profit group of companies and universities,
          makes the MLPerf suites. The submitters review the results of
          each other, and the code of each result is public.
  \end{itemize}
  \source{\emph{Machine Learning Systems}, Volume I, chapter 12.}
\end{frame}

\begin{frame}{MLPerf Tiny: the harness and the run rules}
  \begin{columns}[c]
    \begin{column}{0.44\textwidth}
      \centering
      \begin{tikzpicture}[font=\scriptsize, >={Stealth[length=4pt]},
          b/.style={draw=coolgray, rounded corners=3pt, fill=boxgray,
                    minimum width=1.55cm, minimum height=0.75cm,
                    align=center}]
        \node[b] (h) at (0,0)        {Computer\\with the runner};
        \node[b] (i) at (0,-1.5)     {Isolation\\(IO manager)};
        \node[b, draw=darkcyan] (d) at (2.75,-1.5) {Device\\under test};
        \node[b, draw=cardinalred] (e) at (2.75,0) {Energy\\monitor};
        \draw[<->] (h) -- node[left] {USB} (i);
        \draw[<->] (i) -- node[above] {serial} (d);
        \draw[->, cardinalred, thick] (e) -- node[right, align=left]
          {supply,\\current} (d);
        \draw[<->] (h) -- (e);
      \end{tikzpicture}
      \par\vskip 0.3em
      {\scriptsize The setup for the energy. For the latency and the
       accuracy, the computer talks to the device directly.}
    \end{column}
    \begin{column}{0.54\textwidth}
      \small
      \begin{itemize}
        \item \textbf{Latency:} 5 runs. Each run has 10 seconds or more and
              10 inferences or more. The result is the median of the
              inferences each second.
        \item \textbf{Accuracy:} one inference for each test input, on the
              device.
        \item \textbf{Energy:} the same 5 runs, with the energy monitor. The
              result is the median in $\mu$J for each inference.
        \item \textbf{Window:} the model only. The feature extraction is not
              in the time.
      \end{itemize}
    \end{column}
  \end{columns}
  \vskip 0.3em
  \small
  \begin{itemize}
    \item \textbf{Closed division:} the reference model and dataset.
          Post-training quantization is permitted, new training is not.
    \item \textbf{Open division:} a different model or training is
          permitted. The submitter documents each change.
  \end{itemize}
  \source{Banbury et al., ``MLPerf Tiny Benchmark'', 2021.}
\end{frame}

\begin{frame}{What a standard benchmark does not tell you}
  \small
  \renewcommand{\arraystretch}{1.2}
  \begingroup\centering
  \begin{tabular}{@{}L{0.44\textwidth}L{0.48\textwidth}@{}}
    \toprule
    \textbf{The suite uses} & \textbf{Your product has} \\
    \midrule
    Its reference model               & Your model, with your classes \\
    Its dataset                       & Your users, your sensor, your room \\
    The time of the model only        & Capture, features, and
                                        post-processing \\
    Inferences with no pause          & A duty cycle with sleep \\
    The best software of the submitter & Your runtime and your settings \\
    A laboratory                      & A closed box, in the sun \\
    \bottomrule
  \end{tabular}\par\endgroup
  \vskip 0.4em
  \begin{itemize}
    \item Use a published result to select a chip or a runtime.
    \item Use your own benchmark for the decisions of your product. It has
          the same seven parts.
    \item Part A of the lab: you write these parts as a protocol before the
          first measurement.
  \end{itemize}
  \keypoint{A benchmark is a model of the use, not the use. Say what your
    benchmark does not include.}
\end{frame}

\exerciseframe{read an MLPerf Tiny result table}{%
  \small
  Two boards of one submitter in MLPerf Tiny v1.2, closed division, with the
  same software. A: Cortex-M33 at 160 MHz. B: Cortex-M7 at 280 MHz.
  \vskip 0.3em

  \begin{tabular}{@{}llrrr@{}}

    A & Image classification & 9.003  & 87.0 & 2579.0 $\mu$J \\

    & \textbf{Mean power} \\

  \begin{enumerate}
    \item $1000 / 16.870 = 59.3$ ms and $1000 / 42.009 = 23.8$ ms.
    \item B is 2.5 times as fast. A uses 2.2 times less energy for one
          inference. The image classifier gives almost the same factors:
          2.7 and 2.0.

          5.5 times the power for 2.5 times the speed.
    \item Board A. The limit is 200 ms, and the two boards are fast enough.
          A uses less than one half of the energy for each inference. The
          table does not give the power in the sleep between two
          inferences: Part 3.
    \item The two boards run the same \code{int8} model with the same
          software. In the closed division, the model is the reference
          model.
  \end{enumerate}
