% ------------------------------------------------------------------ why
\begin{frame}{Why an edge device needs security}
  \small
  \begin{itemize}
    \item A device stands in a public place for years. A person can touch
          it, open it, or take it away.
    \item It has a network link, a model, keys, and data of the people near
          it. Each one is worth something to an attacker.
    \item Many identical devices share one weakness. One method attacks all
          of them.
    \item Example: in 2016 the Mirai botnet took control of more than
          600\,000 cameras, recorders, and routers with factory-default
          passwords, and used them for an attack of 1.2 Tbit/s.
  \end{itemize}
  \vskip 0.4em
  \begin{block}{Three goals}
    \textbf{Confidentiality}: only the right people read the data and the
    model. \textbf{Integrity}: nobody changes the data, the model, or the
    commands. \textbf{Availability}: the device works when it is needed.
  \end{block}
  \source{\emph{Machine Learning Systems}, Volume II, chapter ``Security
    \& Privacy'' (Mirai).}
\end{frame}

% ------------------------------------------------------------------ acl
\begin{frame}[fragile]{Passwords and access rules in Mosquitto}
\begin{lstlisting}[style=shell]
mosquitto_passwd -c /etc/mosquitto/passwd g07   # asks for the password of g07
\end{lstlisting}
\begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
# /etc/mosquitto/conf.d/lab.conf
listener 1883
allow_anonymous false
password_file /etc/mosquitto/passwd
acl_file /etc/mosquitto/acl

# /etc/mosquitto/acl: each user only in the tree of its own group
pattern readwrite edgeai/%u/#
\end{lstlisting}
  \small
  \begin{itemize}
    \item \code{\%u} is the user name. The user g07 reads and writes only
          below \code{edgeai/g07/}.
    \item A message to a topic that the rule denies is dropped with no
          error, and a subscription to \code{\#} gets nothing of the other
          groups.
    \item Better: one user for each device, so that a stolen device can be
          removed alone.
  \end{itemize}
  \source{The settings of the experiment of this part; the file names are
    those of a Raspberry Pi.}
\end{frame}

\begin{frame}{Encrypted transport: TLS}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \footnotesize
      \renewcommand{\arraystretch}{1.15}
      \setlength{\tabcolsep}{3pt}
      \begin{tabular}{@{}lrr@{}}
        \toprule
        & \textbf{Plain} & \textbf{TLS} \\
        \midrule
        Connect, median & 2.4 ms & 51.9 ms \\
        Bytes to connect & 36 & 3368 \\
        Bytes of 100 messages & 10\,400 & 12\,600 \\
        One message & 104 bytes & 126 bytes \\
        \bottomrule
      \end{tabular}
    \end{column}
    \begin{column}{0.40\textwidth}
      \small
      An IMU message of 81 bytes of JSON, QoS 0, with a password. A counting
      proxy measured the bytes.
    \end{column}
  \end{columns}
  \vskip 0.6em
  \small
  \begin{itemize}
    \item TLS encrypts each message and checks that nobody changed it. The
          certificate proves the identity of the broker.
    \item The cost is at the start: a handshake with certificates. Keep the
          connection open: each message then costs 22 bytes more.
    \item The ESP32-S3 has AES, SHA, and RSA hardware that makes TLS
          possible on the XIAO (port 8883 by convention).
  \end{itemize}
  \source{Experiment of this part (OpenSSL 3.0.2, x86). ESP32-S3 data sheet
    2.2, section ``Security''.}
\end{frame}

% ------------------------------------------------------------------ keys
\begin{frame}{Keys}
  \small
  \begin{itemize}
    \item \textbf{One key for each device.} A shared password lets one
          stolen device open all others. A key for each device can be
          revoked alone.
    \item \textbf{No default password.} The Mirai botnet used factory
          passwords. A device sets its key at installation.
    \item \textbf{Not in the code repository.} The Day 12 sketch keeps the
          Wi-Fi password in \code{arduino\_secrets.h}, outside the shared
          code. A key in git stays in its history.
    \item \textbf{Protected on the device.} The ESP32-S3 has 4096 bits of
          eFuse memory, an HMAC unit, and an RSA signature unit whose private
          key only the hardware can decrypt. On a Raspberry Pi, a key is a
          file: set its rights (\code{chmod 600}).
    \item \textbf{Change them.} Plan how to replace a key or a certificate
          before it expires.
  \end{itemize}
  \source{ESP32-S3 data sheet 2.2, sections ``Security'', ``HMAC
    Accelerator'', ``RSA Digital Signature Peripheral''.}
\end{frame}

% ------------------------------------------------------------------ theft
\begin{frame}[fragile]{Model theft: the file is the model}
\begin{lstlisting}[style=python, basicstyle=\ttfamily\scriptsize]
interp = tf.lite.Interpreter("mnv2_int8.tflite")   # the Day 7 lab file
interp.allocate_tensors()
# 109 constant tensors with 3 487 830 values, for example:
# shape [1280, 1, 1, 320]: -17, 29, -1, -44, 80, 55, 62, 23, ...
\end{lstlisting}
  \small
  \begin{itemize}
    \item A person with the SD card or the flash of a device has the model:
          the standard interpreter reads each weight.
    \item Encryption at rest: AES-256-GCM of the 4\,107\,968-byte file took
          13.1 ms, its decryption 13.1 ms (one x86 core). A changed byte makes
          the decryption fail.
    \item The weak point is the key: it must be on the device to decrypt.
          The ESP32-S3 has flash encryption in hardware. A person who takes
          the SD card of a Raspberry Pi reads all files that the software
          did not encrypt.
  \end{itemize}
  \source{Experiment of this part (LiteRT interpreter of TensorFlow,
    package cryptography). ESP32-S3 data sheet 2.2.}
\end{frame}

% ------------------------------------------------------------------ integrity
\begin{frame}{Integrity: signed updates and secure boot}
  \begingroup\centering
  \begin{tikzpicture}[font=\scriptsize, >=stealth,
      st/.style={draw=coolgray, rounded corners=2pt, fill=boxgray,
                 align=center, text width=1.75cm, minimum height=0.85cm}]
    \node[st] (a) at (0,0) {Server: SHA-256 of the file};
    \node[st, fill=cardinalred!12] (b) at (2.6,0) {Sign the hash (private
      key)};
    \node[st] (c) at (5.2,0) {Device: SHA-256 again};
    \node[st, fill=darkcyan!15] (d) at (7.8,0) {Check the signature (public
      key)};
    \draw[->] (a) -- (b); \draw[->] (b) -- (c); \draw[->] (c) -- (d);
  \end{tikzpicture}\par\endgroup
  \vskip 0.5em
  \small
  \begin{itemize}
    \item A hash proves that the file did not change in transit. A signature
          also proves who made it: only the holder of the private key can
          sign.
    \item Cost on one x86 core: SHA-256 of the 4\,107\,968-byte file 11.3 ms;
          an Ed25519 signature of 64 bytes, its check 0.14 ms.
    \item Secure boot applies the same check to the program at each start.
          The ESP32-S3 boots only firmware with a valid RSA-PSS signature
          when secure boot is on.
  \end{itemize}
  \source{Experiment of this part. ESP32-S3 data sheet 2.2, section
    ``Secure Boot''.}
\end{frame}

\begin{frame}{Physical access}
  \footnotesize
  \renewcommand{\arraystretch}{1.15}
  \setlength{\tabcolsep}{4pt}
  \begin{tabular}{@{}L{0.30\textwidth}L{0.60\textwidth}@{}}
    \toprule
    \textbf{Attack} & \textbf{Defense} \\
    \midrule
    Read the SD card or the flash & Encrypt the data and the model; keep no
      data that the task does not need \\
    Use the USB or the debug port & Disable the debug port (an eFuse of the
      ESP32-S3); no login with a default password \\
    Load a different program & Secure boot; signed updates \\
    Replace or cover the sensor & Input checks (Day 13); a second sensor;
      a tamper switch \\
    Take the device away & A key for each device that the server can revoke
      \\
    \bottomrule
  \end{tabular}
  \vskip 0.6em
  \small
  \begin{itemize}
    \item With physical access, a network password does not help. The
          defense must be in the hardware and in the boot chain.
    \item A case with a lock and a mount out of reach are also defenses.
  \end{itemize}
  \source{\emph{Machine Learning Systems}, Volume II, chapter ``Security
    \& Privacy'' (physical attacks). ESP32-S3 data sheet 2.2.}
\end{frame}

% ------------------------------------------------------------------ inputs
\begin{frame}{False inputs}
  \small
  \begin{itemize}
    \item A model has no common sense. A printed photo, a sticker, a sound
          from a speaker, or a light can change its output.
    \item On a device, the attacker controls the scene, not the code: the
          attack goes through the sensor.
    \item Defenses that this course built:
          \begin{itemize}
            \item input checks: brightness, sharpness, value range (Day 13);
            \item the decision layer: several windows, hysteresis, a second
                  condition before an action (Day 12);
            \item a second sensor that must agree for an important action.
          \end{itemize}
    \item Poisoned data: a sample that an attacker places in the next
          training set. Check the origin of new data before you train
          (Day 13, the loop back to training).
  \end{itemize}
  \keypoint{Do not let one model output start an action that can hurt
    somebody.}
\end{frame}

  \end{itemize}

% ------------------------------------------------------------------ watchdog
\begin{frame}[fragile]{Reliability: watchdogs}
\begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
[Service]                       # a unit of systemd for the detector
Type=notify
ExecStart=/home/edge/yolo/bin/python monitor_detect.py ...
WatchdogSec=3                   # no ping for 3 s: stop and restart
Restart=on-failure
RestartSec=1
\end{lstlisting}
  \small
  \begin{itemize}
    \item The program sends \code{WATCHDOG=1} to systemd in each loop. A
          program that hangs stops sending, and systemd restarts it.
    \item Test with a program that stopped its pings after 6 s: systemd
          started it again 4.3 s and 4.5 s after the last ping.
    \item The ESP32-S3 has three watchdog timers in hardware. Two of them
          are on during the boot, to recover from a failed start.
  \end{itemize}
  \source{Experiment of this part (\texttt{systemd-run --user} with the same
    settings). ESP32-S3 data sheet 2.2, section ``Watchdog Timers''.}
\end{frame}

% ------------------------------------------------------------------ exercise
\exerciseframe{Threats of the Day 12 system}{%
  The Day 12 system: the XIAO sends keyword events with MQTT to the broker
  on the Raspberry Pi; \code{decide.py} switches the LED of the XIAO; the
  forwarder sends the decisions to the cloud broker. The lab broker is open
  and has no TLS.
  \begin{enumerate}
    \item List at least five threats. For each one, name the point of the
          attack surface and the goal that it harms.
    \item Select two mitigations that remove the largest risk for the least
          work. Say what each one costs.
  \end{enumerate}}

\answerframe{Threats of the Day 12 system}{%
  \footnotesize
  \begin{itemize}
    \item \textbf{False commands}: any client publishes \code{led=1} (broker;
          integrity). \textbf{False events}: any client publishes ``yes''
          (broker; integrity). \textbf{Eavesdropping} on the Wi-Fi
          (radio; confidentiality). \textbf{Flood} of the broker (broker;
          availability). \textbf{Wi-Fi password} read from a stolen XIAO
          (flash; confidentiality). \textbf{Queue and logs} read from the SD
          card (storage; privacy). \textbf{A sound} that the model takes as
          ``yes'' (sensor; integrity).
    \item \textbf{Mitigation 1}: a user for each device and the access rule
          \code{pattern readwrite edgeai/\%u/\#}. It stops the false commands
          and events of the measured test. Cost: a password file and one
          setting on each device.
    \item \textbf{Mitigation 2}: TLS on the broker. It stops the
          eavesdropping. Cost in the test: 51.9 ms to connect and 22 bytes more
          for each message; certificates to manage.
    \item Other good answers: flash encryption on the XIAO, a rate limit,
          a second condition in \code{decide.py}.
  \end{itemize}}
