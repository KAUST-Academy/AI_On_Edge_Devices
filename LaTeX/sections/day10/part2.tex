% ------------------------------------------------------------------ llama.cpp
\begin{frame}[fragile]{llama.cpp: the engine}
  \small
  \begin{itemize}
    \item A C and C++ library with no dependencies. It has NEON kernels for
          Arm and kernels for x86. It reads GGUF files.
    \item You build it, or you download a release. On the Raspberry Pi 5 a
          build with 4 jobs needs about 10 minutes.
    \item No service and no model manager: you download the GGUF file and
          give its path.
  \end{itemize}
  \vskip 0.2em
  \footnotesize
  \renewcommand{\arraystretch}{1.12}
  \begingroup\centering
  \begin{tabular}{@{}L{0.24\textwidth}L{0.68\textwidth}@{}}
    \toprule
    \textbf{Program} & \textbf{Use} \\
    \midrule
    \code{llama-cli}    & A prompt or a chat in the terminal \\
    \code{llama-server} & An HTTP server with an OpenAI-compatible API,
                          port 8080 \\
    \code{llama-bench}  & A benchmark of the prompt rate and the generation
                          rate \\
    \bottomrule
  \end{tabular}\par\endgroup
\begin{lstlisting}[style=shell]
llama-server -m llama3.2-3b-q4_k_m.gguf -c 4096 -t 4 --port 8080
\end{lstlisting}
  \small
  \code{-c}: the context. \code{-t}: the threads. Give the context
  yourself: its default 0 means ``the context of the model file''.
  \source{llama.cpp documentation: README and \code{tools/server}.
    Build time: guide\\ ``SLMs at the Edge'', section 5.2.}
\end{frame}

% ------------------------------------------------------------------ Ollama
\begin{frame}[fragile]{Ollama: a service with a model library}
\begin{lstlisting}[style=shell]
curl -fsSL https://ollama.com/install.sh | sh   # install the service
ollama pull llama3.2:1b                         # download a model
ollama run llama3.2:1b --verbose                # chat, print the rates
ollama ps                                       # loaded models, RAM
\end{lstlisting}
  \small
  \begin{itemize}
    \item A service in Go around the engine of llama.cpp. It starts with
          the board and listens on port 11434.
    \item A model name has a tag: \code{llama3.2:1b} is the 1B model at
          \code{Q8\_0}, \code{llama3.2:3b} the 3B model at
          \code{Q4\_K\_M}.
    \item The service keeps a model in RAM for 5 minutes after the last
          request (\code{keep\_alive}). The next request does not load it
          again.
    \item A REST API for all programs, and a Python library that calls
          this API.
  \end{itemize}
  \keypoint{Ollama is the shortest way from an empty board to a model that
    answers. The lab uses it.}
  \source{Ollama documentation: README, API, and FAQ. Guide ``SLMs at
    the Edge'', section 3.}
\end{frame}

% ------------------------------------------------------------------ LiteRT-LM
\begin{frame}[fragile]{LiteRT-LM}
  \small
  \begin{itemize}
    \item The runtime of Google for language models on LiteRT. LiteRT is
          the new name of TensorFlow Lite (Days 5 to 9).
    \item A \code{.litertlm} file holds the weights (\code{int4} or
          \code{int8}), the tokenizer, and the settings.
    \item On the Raspberry Pi 5 it runs on the CPU with XNNPACK.
    \item The Python API has an \code{Engine}, which loads the model once,
          and a \code{Conversation}, which keeps the KV cache of the chat.
  \end{itemize}
\begin{lstlisting}[style=shell]
uv tool install litert-lm
litert-lm run --from-huggingface-repo=google/gemma-3n-E2B-it-litert-lm \
    gemma-3n-E2B-it-int4 --prompt="What is the capital of France?"
\end{lstlisting}
  \begin{itemize}
    \item Published on the Raspberry Pi 5: Gemma 4 E2B gives about 4 to 8
          tokens/s on the CPU, similar to Ollama with a model of the same
          size.
    \item The same runtime runs on phones, in the browser, and on the
          board. Fewer models than Ollama: about 20.
  \end{itemize}
  \source{LiteRT-LM README. Chapter ``LiteRT-LM'' of
    \emph{Edge AI Engineering}.}
\end{frame}

% ------------------------------------------------------------------ API
\begin{frame}[fragile]{One request to the Ollama API}
\begin{lstlisting}[style=shell]
curl http://localhost:11434/api/generate -d '{
  "model": "llama3.2:1b", "prompt": "Why is the sky blue?", "stream": false,
  "options": {"num_ctx": 2048, "num_thread": 4, "temperature": 0, "seed": 0}
}'
\end{lstlisting}
  \small
  The answer is JSON. It has the text in \code{response} and these
  counters:
  \vskip 0.2em
  \footnotesize
  \renewcommand{\arraystretch}{1.1}
  \begingroup\centering
  \begin{tabular}{@{}L{0.32\textwidth}L{0.60\textwidth}@{}}
    \toprule
    \textbf{Field} & \textbf{Content} \\
    \midrule
    \code{total\_duration}       & The time of the complete request, in ns \\
    \code{load\_duration}        & The time to load the model, in ns \\
    \code{prompt\_eval\_count}   & The tokens of the prompt \\
    \code{prompt\_eval\_duration} & The time of the prompt, in ns \\
    \code{eval\_count}           & The tokens of the answer \\
    \code{eval\_duration}        & The time of the answer, in ns \\
    \bottomrule
  \end{tabular}\par\endgroup
  \vskip 0.3em
  \small
  Generation rate = \code{eval\_count} / \code{eval\_duration} $\times
  10^9$. The prompt rate is the same calculation with the two prompt
  fields.
  \source{Ollama API documentation.}
\end{frame}

% ------------------------------------------------------------------ cold
\begin{frame}{The first request loads the model}
  \small
  Two requests with the same prompt (116 tokens) and 64 tokens of answer,
  \code{llama3.2:1b}, 4 threads:
  \vskip 0.3em
  \renewcommand{\arraystretch}{1.15}
  \begin{tabular}{@{}L{0.30\textwidth}rrrr@{}}
    \toprule
    \textbf{Request} & \textbf{Total} & \textbf{Load} & \textbf{Prompt} &
    \textbf{Answer} \\
    \midrule
    First, after an unload & 7.19 s & 2.95 s & 1.44 s & 2.80 s \\
    Second                 & 4.61 s & 0.58 s & 1.21 s & 2.81 s \\
    \bottomrule
  \end{tabular}
  \vskip 0.4em
  \begin{itemize}
    \item The load was 41 percent of the first request. The file was already
          in the page cache of the operating system.
    \item From a microSD card, a model of 1.5 GB loads in 10 to 20 s; from
          an NVMe SSD in 2 to 4 s (published).
    \item A benchmark: send one request first and do not count it, as the
          warm-up of Day 9.
    \item A product: keep the model loaded. \code{keep\_alive}: $-1$ keeps it,
          0 unloads it at once.
  \end{itemize}
  \source{Experiment of this course: Ollama 0.32.6, x86 work computer, CPU
    only.\\ Load times of the board: chapter ``LiteRT-LM'' of \emph{Edge AI
    Engineering}.}
\end{frame}

\begin{frame}{Set the options yourself}
  \footnotesize
  \renewcommand{\arraystretch}{1.2}
  \begingroup\centering
  \begin{tabular}{@{}L{0.16\textwidth}L{0.18\textwidth}L{0.20\textwidth}L{0.34\textwidth}@{}}
    \toprule
    \textbf{Setting} & \textbf{Ollama} & \textbf{llama.cpp} &
    \textbf{Documented default} \\
    \midrule
    Context      & \code{num\_ctx}     & \code{-c}     & Ollama: 4096 in the FAQ,
                                                       2048 in the Modelfile
                                                       table. llama.cpp: the
                                                       context of the model. \\
    Threads      & \code{num\_thread}  & \code{-t}, \code{-tb} & Set by the
                                                       runtime \\
    Answer length & \code{num\_predict} & \code{-n}    & Ollama: $-1$, no limit \\
    Randomness   & \code{temperature}, \code{seed} & \code{--temp},
                   \code{--seed} & Ollama: temperature 0.8 \\
    Stay in RAM  & \code{keep\_alive}  & the server runs & Ollama: 5 minutes \\
    \bottomrule
  \end{tabular}\par\endgroup
  \vskip 0.5em
  \small
  \begin{itemize}
    \item Two pages of one documentation give two defaults for the
          context. Do not depend on a default: give the value in the
          request.
    \item The context of Llama 3.2 is 131\,072 tokens. With the default of
          llama.cpp, the KV cache of the 3B model needs 15 GB (Part 1).
    \item Temperature 0 and a fixed seed give the same answer for the same
          prompt: use them for a benchmark.
  \end{itemize}
  \source{Ollama documentation: FAQ, Modelfile, and API.\\ llama.cpp
    documentation of the CLI and the server.}
\end{frame}

% ------------------------------------------------------------------ context
\begin{frame}{Measured: the context costs RAM}
  \small
  \code{llama3.2:3b}, the RAM of the loaded model in \code{ollama ps}, for
  three values of \code{num\_ctx}:
  \vskip 0.3em
  \renewcommand{\arraystretch}{1.15}
  \begin{tabular}{@{}rrrrr@{}}
    \toprule
    \textbf{\code{num\_ctx}} & \textbf{KV cache (Part 1)} & \textbf{File +
    KV cache} & \textbf{Measured} & \textbf{Difference} \\
    \midrule
    2048    & 0.235 GB & 2.25 GB & 2.32 GB & 0.07 GB \\
    8192    & 0.940 GB & 2.96 GB & 3.12 GB & 0.16 GB \\
    32\,768 & 3.758 GB & 5.78 GB & 5.99 GB & 0.21 GB \\
    \bottomrule
  \end{tabular}
  \vskip 0.4em
  \begin{itemize}
    \item The formula of Part 1 predicts the measured RAM. The difference
          is the buffers of the runtime: they also grow a little with the
          context.
    \item From 2048 to 32\,768 tokens, the RAM grows by 119\,262 bytes for
          each token. The formula gives 114\,688: 4 percent less.
    \item On the 8 GB board, 32\,768 tokens leave little RAM for the system
          and your program.
  \end{itemize}
  \keypoint{Calculate the RAM before the lab, then check it with
    \code{ollama ps} and \code{free}.}
  \source{Experiment of this course: Ollama 0.32.6, x86 work computer, CPU
    only.\\ The RAM of the model does not depend on the computer.}
\end{frame}

  \begin{itemize}
    \item The RAM is a hard limit. If the board swaps, the rate falls: use
          a smaller model or a smaller level.
    \item Take the instruct or chat version, not the base model.
    \item A reasoning model writes many tokens before the answer. On the

% -------------------------------------------------------------------- exercise
\exerciseframe{select a model for three applications}{%
  \small
  Use the candidates of the guide and the budget of 5.3 GB on a Raspberry
  Pi 5 with 8 GB. For each application, select a model and a quantization
  level. Give the rate that you need and the RAM.
  \begin{enumerate}
    \item \textbf{Home assistant.} A person speaks a short question and
          reads the answer on a screen. A speech model and a voice model
          run on the same board.
    \item \textbf{Night report.} Each night the board summarizes the log of
          20 sensors: about 6000 tokens of text. Nobody waits for it.
    \item \textbf{Robot commands.} A command such as ``move to the red box
          and stop'' becomes a JSON object with 3 fields. The robot must
          start to move in less than 5 seconds.
  \end{enumerate}
  For application 2, calculate the KV cache of Llama 3.2 3B for a context
  of 8192 tokens (Part 1: 114\,688 bytes for each token).
}

\answerframe{select a model for three applications}{%
  \small
  \begin{enumerate}
    \item Qwen3.5 0.8B at \code{Q8\_0}, or Llama 3.2 1B. A chat needs 8 to
          10 tokens/s, and these models give 8 to 25. About 1 GB leaves RAM
          for the speech and voice models.
    \item Llama 3.2 3B at \code{Q4\_K\_M}: 2 to 5 tokens/s is enough for a
          job at night, and the summary is better. KV cache: $114\,688
          \times 8192 = 0.94$ GB. Total $2.2 + 0.94 = 3.14$ GB, below 5.3
          GB. A prompt of 6000 tokens takes minutes: this is acceptable
          at night.
    \item A model of about 1B at \code{Q5\_K\_M} or \code{Q8\_0}, with
          temperature 0, a JSON schema (Part 3), and a short
          \code{num\_predict}. The prompt is short and the answer has about
          30 tokens: at about 10 tokens/s, 3 s.
  \end{enumerate}
  \keypoint{The application sets the rate and the context. Then select the
    smallest model that passes your test prompts.}
}
