% Learning outcomes of the module. Module TH-3 of Section 6.7 of the
% syllabus: "ML workflow and life cycle". The syllabus gives no list of
% outcomes for this module, so the three outcomes come from its description.
\begin{frame}{Learning outcomes}
  After this module you can:
  \begin{itemize}
    \item Name the six stages of an ML project, from the problem definition
          to the monitoring, and the output of each stage.
    \item Explain why a deployment constraint that nobody writes down comes
          back later as a redesign.
    \item Use the feedback of the monitoring stage to decide which earlier
          stage to change.
  \end{itemize}
\end{frame}
