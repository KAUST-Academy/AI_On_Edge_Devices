\begin{frame}{Quantize a model that is trained}
  \begingroup\centering
  \begin{tikzpicture}[font=\footnotesize, >={Stealth[length=5pt]},

                minimum width=2.3cm, minimum height=1.0cm, align=center}]
    \node[b] (m) at (0,0.7)  {Trained model\\\code{float32}};
    \node[b] (d) at (0,-0.7) {Calibration\\data};
    \node[b, draw=cardinalred] (c) at (3.6,0) {Converter};
    \node[b] (q) at (7.2,0)  {LiteRT model\\\code{int8}};
    \draw[->] (m.east) -- (c.west);
    \draw[->] (d.east) -- (c.west);
    \draw[->] (c) -- (q);
    \node[font=\scriptsize, text=coolgray] at (3.6,-0.8) {no training};
  \end{tikzpicture}\par\endgroup
  \vskip 0.5em
  \begin{itemize}
    \item Post-training quantization (PTQ) changes the format of a model
          after the training.
    \item It needs no training loop, no labels, and no loss function. It
          needs some seconds.
    \item The converter must know the range of each tensor. It then
          calculates the scale and the zero point of Part 1.
  \end{itemize}
  \vskip 0.2em
  \keypoint{Always try post-training quantization first. Part 3 gives the
    method for a model that loses too much accuracy.}
\end{frame}

\begin{frame}{Weights and activations are different problems}
  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      \begin{block}{Weights}
        \begin{itemize}
          \item They are in the model file.
          \item They do not change after the training.
          \item The converter reads the smallest and the largest weight of
                each tensor. It needs no data.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{block}{Activations}
        \begin{itemize}
          \item They exist only when the model runs.
          \item They depend on the input.
          \item The converter must run the model with real inputs and
                record the range of each activation.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
  \vskip 0.6em
  \begin{itemize}
    \item This difference gives two methods: one quantizes only the
          weights, the other quantizes the weights and the activations.
  \end{itemize}
  \keypoint{The range of an activation is a property of the model and of
    its data together.}
\end{frame}

% -------------------------------------------------------------- dynamic range
\begin{frame}[fragile]{Dynamic range quantization}
\begin{lstlisting}[style=python]
converter = tf.lite.TFLiteConverter.from_keras_model(model)
converter.optimizations = [tf.lite.Optimize.DEFAULT]
tflite_model = converter.convert()
\end{lstlisting}
  \begin{itemize}
    \item One line more than the float conversion. No data is necessary.
    \item The weights are \code{int8} in the file. The input, the output,
          and the activations stay \code{float32}.
    \item At run time, the kernel finds the range of each activation for
          each input. From this comes the name ``dynamic range''.
  \end{itemize}
  \vskip 0.3em
  \small
  \renewcommand{\arraystretch}{1.25}
  \begingroup\centering
  \begin{tabular}{@{}L{0.30\textwidth}L{0.62\textwidth}@{}}
    \toprule
    Good for & A phone or a Raspberry Pi: the file is smaller, and the
               sketch or the program needs no change \\
    Not good for & A microcontroller: the arena does not become smaller,
               and the kernels still use float arithmetic \\
    \bottomrule
  \end{tabular}\par\endgroup
  \source{The arena of the dynamic range model is equal to the arena of the
    float model: 1232 bytes for the feature model of Day 3, in a 32-bit
    build of TensorFlow Lite Micro on the work computer.}
\end{frame}

% --------------------------------------------------------------- full integer
\begin{frame}[fragile]{Full integer quantization}
\begin{lstlisting}[style=python]
def representative_dataset():
    for i in range(200):
        yield [x_calibration[i:i + 1]]         # one input, with batch size 1

converter = tf.lite.TFLiteConverter.from_keras_model(model)
converter.optimizations = [tf.lite.Optimize.DEFAULT]
converter.representative_dataset = representative_dataset
converter.target_spec.supported_ops = [tf.lite.OpsSet.TFLITE_BUILTINS_INT8]
converter.inference_input_type = tf.int8
converter.inference_output_type = tf.int8
tflite_model = converter.convert()
\end{lstlisting}
  \small
  \begin{itemize}
    \item \code{representative\_dataset}: the converter runs the model with
          these inputs and records the range of each activation.
    \item \code{TFLITE\_BUILTINS\_INT8}: each operator must have an
          \code{int8} kernel. If one operator has none, the conversion
          stops with an error.
    \item \code{inference\_input\_type}: the model takes \code{int8}
          input. The sketch quantizes the input, as Part 1 showed.
  \end{itemize}
\end{frame}

\begin{frame}{Calibration: record the range of each activation}
  \begingroup\centering
  \begin{tikzpicture}[font=\footnotesize, >={Stealth[length=5pt]},

      o/.style={draw=cardinalred, rounded corners=3pt, fill=white,
                minimum width=1.9cm, minimum height=0.7cm, align=center,
                font=\scriptsize}]
    \node[b] (i) at (0,0)   {Inputs};
    \node[b] (a) at (2.6,0) {Layer 1};
    \node[b] (c) at (5.2,0) {Layer 2};
    \node[b] (e) at (7.8,0) {Layer 3};
    \draw[->] (i) -- (a); \draw[->] (a) -- (c); \draw[->] (c) -- (e);
    \node[o] (o0) at (1.3,-1.3) {min $-0.87$\\max 1.74};
    \node[o] (o1) at (3.9,-1.3) {min 0.00\\max 5.31};
    \node[o] (o2) at (6.5,-1.3) {min 0.00\\max 7.92};
    \draw[->, cardinalred] (1.3,0) -- (o0);
    \draw[->, cardinalred] (3.9,0) -- (o1);
    \draw[->, cardinalred] (6.5,0) -- (o2);
  \end{tikzpicture}\par\endgroup
  \vskip 0.4em
  \begin{enumerate}
    \item The converter puts an observer at each activation tensor.
    \item It runs the float model for each input of the calibration set.
    \item Each observer keeps the smallest and the largest value that it
          saw.
    \item From each range, the converter calculates one scale and one zero
          point.
  \end{enumerate}
  \vskip 0.2em
  \keypoint{The calibration is a measurement of the model with your data.
    It trains nothing and changes no weight.}
\end{frame}

\begin{frame}{The representative dataset}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \begin{block}{What it must be}
        \begin{itemize}
          \item Inputs of the same type as the inputs on the device.
          \item With the same pre-processing as the training.
          \item From all classes and all conditions.
          \item 100 to 500 inputs are usually enough.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{What it does not need}
        \begin{itemize}
          \item Labels.
          \item Many inputs: the range of a tensor is stable after some
                hundreds.
          \item The test set. Take the inputs from the training set.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
  \vskip 0.6em
  \begin{itemize}
    \item A range that is too small cuts values on the device: clipping.
    \item A range that is too large makes the step large for all values:
          rounding error.
  \end{itemize}
  \keypoint{The calibration set decides the range. A wrong calibration set
    gives a wrong model, and the conversion shows no error.}
\end{frame}

\begin{frame}{Calibration methods}
  \small
  \renewcommand{\arraystretch}{1.3}
  \begingroup\centering
  \begin{tabular}{@{}L{0.20\textwidth}L{0.40\textwidth}L{0.30\textwidth}@{}}
    \toprule
    \textbf{Method} & \textbf{The range is} & \textbf{Weak point} \\
    \midrule
    Minimum and maximum & The smallest and the largest value that the
        observer saw & One extreme value makes the range large \\
    Moving average & The mean of the minimum and of the maximum over the
        batches & Depends on the order and the size of the batches \\
    Percentile & The range that holds 99.9 percent of the values
        & Cuts the rare, large values \\
    Minimum error & The range that gives the smallest difference between
        the float values and the quantized values
        & Needs more calculation \\
    \bottomrule
  \end{tabular}\par\endgroup
  \vskip 0.6em
  \begin{itemize}
    \item The LiteRT converter records the minimum and the maximum.
    \item Other tools let you select the method. Day 7 uses a second tool.
  \end{itemize}
  \source{Method list: \emph{Machine Learning Systems}, Volume I,
    chapter 10; Nagel et al., ``A White Paper on Neural Network
    Quantization'', 2021.}
\end{frame}

\begin{frame}{Per-tensor quantization of a convolution}
  \begingroup\centering
  \begin{tikzpicture}[font=\scriptsize, x=0.85cm, y=0.55cm]
    \foreach \y/\n/\w/\c in {3/filter 1/4.0/darkcyan, 2/filter 2/0.4/treegreen,
        1/filter 3/1.6/cardinalred} {
      \node[anchor=east] at (-4.3,\y) {\n};
      \fill[\c!45] (-\w,\y-0.3) rectangle (\w,\y+0.3);
    }
    \draw[coolgray] (0,0.4) -- (0,3.6);
    \draw[->, coolgray] (-4.2,0.2) -- (4.4,0.2) node[right] {weight};
    \foreach \x/\l in {-4/$-1.0$, 0/0, 4/1.0} { \node[below] at (\x,0.2) {\l}; }
    \node[anchor=west] at (4.5,3) {$\pm 1.0$};
    \node[anchor=west] at (4.5,2) {$\pm 0.1$};
    \node[anchor=west] at (4.5,1) {$\pm 0.4$};
  \end{tikzpicture}\par\endgroup
  \vskip 0.3em
  \begin{itemize}
    \item A convolution has one filter for each output channel. The
          weights of different filters can have very different sizes.
    \item Per-tensor: one scale for all filters. The largest filter
          decides: $S = 1.0 / 127 = 0.00787$.
    \item Filter 2 has weights up to 0.1. With this scale, its integers
          are between $-13$ and 13. It uses 27 of the 255 levels.
  \end{itemize}
  \vskip 0.2em
  \keypoint{With one scale for the tensor, a filter with small weights
    has a large relative error.}
\end{frame}

\begin{frame}{Per-channel quantization}
  \small
  \renewcommand{\arraystretch}{1.25}
  \setlength{\tabcolsep}{4pt}
  \begin{tabular}{@{}lrrrr@{}}
    \toprule
    & \textbf{Largest} & \textbf{Per-tensor} & \textbf{Per-channel}
    & \textbf{Levels} \\
    & \textbf{weight} & \textbf{scale} & \textbf{scale}
    & \textbf{in use} \\
    \midrule
    Filter 1 & 1.0 & 0.00787 & 0.00787 & 255 and 255 \\
    Filter 2 & 0.1 & 0.00787 & 0.00079 & 27 and 255 \\
    Filter 3 & 0.4 & 0.00787 & 0.00315 & 103 and 255 \\
    \bottomrule
  \end{tabular}
  \vskip 0.6em
  \begin{itemize}
    \item Per-channel: each filter has its own scale. Each filter then
          uses all levels.
    \item The cost is small: one scale for each output channel in the
          file, and one multiplier $M$ for each channel in the kernel.
    \item The zero point of the weights stays 0 for all channels.
    \item The activations stay per-tensor: one scale and one zero point
          for each activation tensor.
  \end{itemize}
  \vskip 0.2em
  \keypoint{The LiteRT converter uses per-channel scales for the weights of
    convolution layers. You see the list of the scales in Netron.}
  \source{The last column gives the levels with the per-tensor scale and
    with the per-channel scale. Rules: \texttt{int8} quantization
    specification of LiteRT.}
\end{frame}

% ------------------------------------------------------------------ BN folding
\begin{frame}{Batch normalization folding}
  \begingroup\centering
  \begin{tikzpicture}[font=\footnotesize, >={Stealth[length=5pt]},
      b/.style={draw=coolgray, rounded corners=3pt, fill=boxgray,
                minimum width=1.9cm, minimum height=0.8cm, align=center}]
    \node[anchor=east, text=coolgray] at (-1.3,0) {Training};
    \node[anchor=east, text=coolgray] at (-1.3,-1.3) {Inference};
    \node[b] (a) at (0,0)   {Convolution\\$w$, $b$};
    \node[b] (c) at (2.8,0) {Batch norm\\$\gamma$, $\beta$, $\mu$, $\sigma$};
    \node[b] (d) at (5.6,0) {ReLU};
    \draw[->] (a) -- (c); \draw[->] (c) -- (d);
    \node[b, draw=cardinalred, minimum width=4.7cm] (e) at (1.4,-1.3)
      {Convolution with $w'$, $b'$};
    \node[b] (f) at (5.6,-1.3) {ReLU};
    \draw[->] (e) -- (f);
  \end{tikzpicture}\par\endgroup
  \vskip 0.2em
  \[
    w' = w \cdot \frac{\gamma}{\sigma} \qquad\qquad
    b' = \beta + (b - \mu) \cdot \frac{\gamma}{\sigma}
  \]
  \small
  \begin{itemize}
    \item At inference, a batch normalization is a multiplication and an
          addition with constants, for each channel.
    \item The converter puts these constants into the weights and the
          bias of the convolution before it. The layer disappears.
    \item The folding must come before the quantization. The device runs
          $w'$, so the range of $w'$ decides the scale.
    \item The factor $\gamma / \sigma$ is different for each channel. So
          the folded filters have very different ranges. This is the
          main reason for per-channel scales.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Check the quantized model on the laptop}
\begin{lstlisting}[style=python]
interpreter = Interpreter(model_content=tflite_model)
interpreter.allocate_tensors()
inp = interpreter.get_input_details()[0]
out = interpreter.get_output_details()[0]
S, Z = inp["quantization"]                   # scale and zero point

correct = 0
for x, label in zip(x_test, y_test):
    q = np.clip(np.round(x / S) + Z, -128, 127).astype(np.int8)
    interpreter.set_tensor(inp["index"], q[None, :])
    interpreter.invoke()
    correct += interpreter.get_tensor(out["index"])[0].argmax() == label
print(correct / len(x_test))
\end{lstlisting}
  \small
  \begin{itemize}
    \item The LiteRT interpreter on the laptop uses the same integer
          arithmetic as the device.
    \item Compare the accuracy of the float model and of the \code{int8}
          model on the same test set.
    \item Count the input values that the clamp changes. Many clipped
          values show a calibration problem.
  \end{itemize}
\end{frame}

\begin{frame}{The steps of post-training quantization}
  \begingroup\centering
  \begin{tikzpicture}[font=\footnotesize, >={Stealth[length=5pt]},
      b/.style={draw=coolgray, rounded corners=3pt, fill=boxgray,
                minimum width=1.75cm, minimum height=1.0cm, align=center}]
    \node[b] (a) at (0,0)    {Float\\baseline};
    \node[b] (b) at (2.15,0) {Calibration\\set};
    \node[b, draw=cardinalred] (c) at (4.3,0) {Convert\\to \code{int8}};
    \node[b] (d) at (6.45,0) {Test on the\\laptop};
    \node[b] (e) at (8.6,0)  {Test on\\the board};
    \foreach \a/\b in {a/b, b/c, c/d, d/e} { \draw[->] (\a) -- (\b); }
  \end{tikzpicture}\par\endgroup
  \vskip 0.6em
  \begin{enumerate}
    \item \textbf{Float baseline.} Measure the accuracy, the file size, and
          the latency of the float model.
    \item \textbf{Calibration set.} Take 100 to 500 inputs from the
          training set, from all classes.
    \item \textbf{Convert.} Full integer quantization with \code{int8}
          input and output.
    \item \textbf{Test on the laptop.} The same test set as for the float
          model. Compare.
    \item \textbf{Test on the board.} Flash, arena, latency, and the class
          for real inputs.
  \end{enumerate}
  \keypoint{Each number of the \code{int8} model needs the same number of
    the float model next to it.}
\end{frame}

\answerframe{a calibration set with one class}{%
  \footnotesize
  \begin{enumerate}
    \item The range of the class ``idle'': about $[0.00, 1.01]$. The scale
          is $1.01 / 255 = 0.0040$.
    \item The two values are outside the range. The clamp changes $-0.87$
          to 0.00 and 1.74 to 1.01. The model receives a window that the
          sensor did not measure.
    \item ``idle'': no change. The motion classes: lower, because a part of
          their values is cut.
    \item Count the clipped input values on the test set, and compare the
          accuracy of each class with the float model.
  \end{enumerate}
