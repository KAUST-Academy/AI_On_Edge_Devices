% ----------------------------------------------------------------- the idea
\begin{frame}{A new model in place of a smaller copy}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Pruning (Part 1)}
        \begin{itemize}
          \item Starts from the trained model and removes parts.
          \item The result has the same layer types and the same
                connections.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Distillation (this part)}
        \begin{itemize}
          \item Trains a new model that you design for the device.
          \item The large model helps in the training. It does not go to
                the device.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
  \vskip 0.5em
  \begin{itemize}
    \item The large model is the \textbf{teacher}. The small model is the
          \textbf{student}.
    \item The student can have a different depth, a different width, and
          different layer types.
    \item The result is a normal dense model. Each runtime runs it with no
          change.
  \end{itemize}
  \keypoint{Distillation moves knowledge from one model to a different
    model. It does not move weights.}
\end{frame}

\begin{frame}{The temperature}
  \[
    p_i = \frac{\exp(z_i / T)}{\sum_j \exp(z_j / T)}
  \]
  \small
  \renewcommand{\arraystretch}{1.2}
  \begin{tabular}{@{}L{0.14\textwidth}rrrrrr@{}}
    \toprule
    & \textbf{Shirt} & \textbf{Pullover} & \textbf{T-shirt} & \textbf{Coat}
    & \textbf{Trouser} & \textbf{Others} \\
    \midrule
    $T = 1$ & 0.909 & 0.070 & 0.011 & 0.010 & 0.000 & 0.000 \\
    $T = 2$ & 0.658 & 0.183 & 0.072 & 0.069 & 0.009 & 0.008 \\
    $T = 4$ & 0.403 & 0.212 & 0.134 & 0.131 & 0.048 & 0.072 \\
    $T = 8$ & 0.250 & 0.181 & 0.144 & 0.142 & 0.086 & 0.196 \\
    \bottomrule
  \end{tabular}
  \vskip 0.5em
  \begin{itemize}
    \item $z_i$ is the logit of class $i$: the output before the softmax.
          $T = 1$ is the normal softmax.
    \item A higher $T$ makes the distribution flatter. The order of the
          classes stays, and the small values become visible.
  \end{itemize}
  \source{The same test image. The column ``Others'' is the sum of the 5
    other classes.}
\end{frame}

\begin{frame}{The distillation loss}
  \[
    L = \alpha \cdot L_{\text{hard}} + (1 - \alpha) \cdot T^2 \cdot
        L_{\text{soft}}
  \]
  \small
  \renewcommand{\arraystretch}{1.3}
  \begin{tabular}{@{}L{0.12\textwidth}L{0.80\textwidth}@{}}
    \toprule
    $L_{\text{hard}}$ & The cross-entropy between the label and the output
        of the student at $T = 1$. It is the normal loss. \\
    $L_{\text{soft}}$ & The difference between the output of the teacher and
        the output of the student, the two at the same temperature $T$.
        The measure is the Kullback-Leibler divergence. \\
    $\alpha$ & The weight of the label, from 0 to 1. A typical value is
        0.1: the teacher counts most. \\
    $T^2$ & A correction. The gradient of the soft part becomes smaller by
        the factor $1/T^2$. The factor $T^2$ keeps the two parts in
        balance. \\
    \bottomrule
  \end{tabular}
  \vskip 0.5em
  \begin{itemize}
    \item The temperature is a tool of the training. On the device, the
          student uses the normal softmax, or no softmax.
  \end{itemize}
  \source{Hinton et al., ``Distilling the Knowledge in a Neural Network'',
    2015; \emph{Machine Learning Systems}, Volume I, chapter 10.}
\end{frame}

\begin{frame}[fragile]{The loss in Keras}
\begin{lstlisting}[style=python]
def distillation_loss(T, alpha):
    def loss(y_true, z_student):
        labels = ops.cast(y_true[:, 0], "int32")    # the label
        z_teacher = y_true[:, 1:]                   # the teacher logits
        hard = keras.losses.sparse_categorical_crossentropy(
            labels, z_student, from_logits=True)
        p_teacher = ops.softmax(z_teacher / T)
        log_p_student = ops.log_softmax(z_student / T)
        soft = ops.sum(p_teacher * (ops.log(p_teacher + 1e-9)
                                    - log_p_student), axis=-1)
        return alpha * hard + (1 - alpha) * T * T * soft
    return loss
\end{lstlisting}
  \small
  \begin{itemize}
    \item The teacher runs one time over the training set. Its logits go
          into the target array, next to the label.
    \item The student has no softmax layer in the training: it gives
          logits.
    \item The training loop is the normal loop: \code{student.fit(x, y)}.
  \end{itemize}
\end{frame}

\begin{frame}{The limits of distillation}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{The student}
        \begin{itemize}
          \item A student that is too small cannot hold the function of the
                teacher. The gain becomes small.
          \item A very large gap between teacher and student can make the
                result worse.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{The teacher}
        \begin{itemize}
          \item The student copies the errors of the teacher also.
          \item A teacher that is too sure gives soft targets near the
                labels. They carry little.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
  \vskip 0.5em
  \begin{itemize}
    \item The data of the distillation must be like the data of the device.
          The student learns the teacher only where it sees examples.
    \item The values of $T$ and $\alpha$ come from experiments. No rule
          gives them.
  \end{itemize}
  \keypoint{Distillation is not a compression with no loss. The student is
    limited by its own size, by the teacher, and by the data.}
  \source{\emph{Machine Learning Systems}, Volume I, chapter 10.}
\end{frame}

          not only its output.

          adapter layer, for example a convolution of $1 \times 1$, makes

    \item The loss is the mean squared difference between the two feature

    \item It helps most for deep students and for tasks with a small
          output, for example two classes: the output alone says little.

\begin{frame}{Low-rank factorization: the idea}
  \begingroup\centering
  \begin{tikzpicture}[font=\footnotesize, >={Stealth[length=5pt]}]
    \draw[coolgray, fill=darkcyan!35] (0,0) rectangle (2.4,2.4);
    \node at (1.2,1.2) {$W$};
    \node[below, font=\scriptsize] at (1.2,0) {$n$ columns};
    \node[left, font=\scriptsize] at (0,1.2) {$m$ rows};
    \node at (3.1,1.2) {$\approx$};
    \draw[coolgray, fill=cardinalred!35] (3.8,0) rectangle (4.5,2.4);
    \node at (4.15,1.2) {$U$};
    \node[below, font=\scriptsize] at (4.15,0) {$k$};
    \node at (4.9,1.2) {$\cdot$};
    \draw[coolgray, fill=cardinalred!35] (5.3,1.7) rectangle (7.7,2.4);
    \node at (6.5,2.05) {$V^{T}$};
    \node[below, font=\scriptsize] at (6.5,1.7) {$n$ columns, $k$ rows};
  \end{tikzpicture}\par\endgroup
  \vskip 0.3em
  \[
    W \approx U \, V^{T} \qquad m \cdot n \ \text{values become} \
    k \cdot (m + n) \ \text{values}
  \]
  \begin{itemize}
    \item A weight matrix with $m$ rows and $n$ columns becomes the product
          of two thin matrices. $k$ is the rank.
    \item The layer $y = W x$ becomes two layers:
          $y = U \, (V^{T} x)$. The device never calculates $W$.
    \item The method saves values only when
          $k < \dfrac{m \cdot n}{m + n}$.
  \end{itemize}
  \source{\emph{Machine Learning Systems}, Volume I, chapter 10.}
\end{frame}

\begin{frame}{How to find the two factors}
  \begin{columns}[c]
    \begin{column}{0.42\textwidth}
      \centering
      \begin{tikzpicture}[font=\scriptsize, x=0.052cm, y=0.24cm]
        \draw[->, coolgray] (0,0) -- (70,0);
        \node[coolgray] at (35,-2.6) {number of the part};
        \draw[->, coolgray] (0,0) -- (0,12.5);
        \node[anchor=west, coolgray] at (1,12.6) {singular value};
        \foreach \x in {16,32,64} {
          \draw[coolgray] (\x,0) -- (\x,-0.3) node[below] {\x};
        }
        \foreach \y in {5,10} {
          \draw[coolgray] (0,\y) -- (-1,\y) node[left] {\y};
        }
        \draw[cardinalred, line width=1pt] plot coordinates {(1,11.17) (4,8.59) (7,7.40) (10,6.45) (13,5.62)
          (16,5.10) (19,4.51) (22,4.10) (25,3.73) (28,3.47) (31,3.09)
          (34,2.89) (37,2.58) (40,2.41) (43,2.20) (46,2.06) (49,1.85)
          (52,1.68) (55,1.49) (58,1.30) (61,1.11) (64,0.71)};
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.56\textwidth}
      \small
      \begin{itemize}
        \item The singular value decomposition (SVD) writes $W$ as a sum
              of parts. Each part has a singular value: its weight in the
              sum.
        \item Keep the $k$ parts with the largest singular values. This is
              the best approximation of $W$ with the rank $k$.
        \item If the singular values fall fast, a small $k$ is enough.
      \end{itemize}
    \end{column}
  \end{columns}
  \vskip 0.4em
  \begin{itemize}
    \item The figure shows the third convolution of the example model as a
          matrix of $288 \times 64$. Its singular values fall slowly: from
          11.2 to 0.7.
  \end{itemize}
  \source{Singular values: the example model of Part 1. Method: the
    Eckart-Young theorem.}
\end{frame}

    \item A small layer of a small model is not low rank: each of its

  \end{itemize}

\exerciseframe{soft targets at two temperatures}{%
  \small
  A teacher gives these logits for one image:
  \vskip 0.2em
  {\footnotesize
  \renewcommand{\arraystretch}{1.1}
  \begin{tabular}{@{}lrrrr@{}}
    \toprule
    \textbf{Class} & Shirt & Pullover & Coat & Sandal \\
    \midrule
    \textbf{Logit $z$} & 4.0 & 2.0 & 0.0 & $-2.0$ \\
    \bottomrule
  \end{tabular}}
  \vskip 0.2em
  \[
    p_i = \frac{\exp(z_i / T)}{\sum_j \exp(z_j / T)}
  \]
  Values: $e^{4} = 54.60$, $e^{2} = 7.39$, $e^{1} = 2.72$,
  $e^{0.5} = 1.65$, $e^{0} = 1$, $e^{-0.5} = 0.61$, $e^{-2} = 0.14$.
  \begin{enumerate}
    \item Calculate the four probabilities for $T = 1$.
    \item Calculate the four probabilities for $T = 4$.
    \item Compare the classes ``Coat'' and ``Sandal'' in the two results.
          What does the student learn at $T = 4$ that it cannot learn at
          $T = 1$?
    \item What is the result for a very large $T$? Is this useful?
  \end{enumerate}
  Work with your neighbour for 8 minutes.
}

\answerframe{soft targets at two temperatures}{%
  \small
  \begin{enumerate}
    \item $T = 1$: the sum is $54.60 + 7.39 + 1.00 + 0.14 = 63.13$.
    \item $T = 4$: the logits become 1.0, 0.5, 0.0, and $-0.5$. The sum is
          $2.72 + 1.65 + 1.00 + 0.61 = 5.98$.
          \vskip 0.2em
          {\footnotesize
          \renewcommand{\arraystretch}{1.05}
          \begin{tabular}{@{}lrrrr@{}}
            \toprule
            & \textbf{Shirt} & \textbf{Pullover} & \textbf{Coat}
            & \textbf{Sandal} \\
            \midrule
            $T = 1$ & 0.865 & 0.117 & 0.016 & 0.002 \\
            $T = 4$ & 0.455 & 0.276 & 0.167 & 0.102 \\
            \bottomrule
          \end{tabular}}
    \item At $T = 1$, the two classes have 0.016 and 0.002: almost no
          signal in the loss. At $T = 4$, they have 0.167 and 0.102. The
          student now learns that a shirt is more like a coat than like a
          sandal. The order of the classes is the same in the two rows.
    \item Each class gets 0.25. All information is lost: the target is the
          same for each image. A very high temperature is not useful.
  \end{enumerate}
}
