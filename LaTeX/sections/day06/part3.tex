\begin{frame}{Design for the device from the start}
  \small
  \renewcommand{\arraystretch}{1.3}
  \begin{tabular}{@{}L{0.26\textwidth}L{0.42\textwidth}L{0.22\textwidth}@{}}
    \toprule
    \textbf{The measured problem} & \textbf{The answer in the architecture}
    & \textbf{Example} \\
    \midrule
    The complete model is too large & Scale the depth, the width, and the

    \bottomrule
  \end{tabular}

  \begin{itemize}
    \item Pruning, distillation, and quantization repair a model that

\begin{frame}{Where the cost of a CNN comes from}
  \[
    \text{MACs of one convolution} = H \cdot W \cdot k^2 \cdot C_{in} \cdot
    C_{out}
  \]
  \small
  \renewcommand{\arraystretch}{1.3}
  \begin{tabular}{@{}L{0.20\textwidth}L{0.34\textwidth}L{0.36\textwidth}@{}}
    \toprule
    \textbf{Control} & \textbf{Changes} & \textbf{Effect on the MACs} \\
    \midrule
    Image size   & $H$ and $W$ of each feature map & With the square of the
        side \\
    Width        & $C_{in}$ and $C_{out}$ of each layer & With the square of
        the width \\
    Depth        & The number of layers & With the depth \\
    Layer type   & The factor $k^2 \cdot C_{in} \cdot C_{out}$
                 & Depthwise and pointwise: about 8 times fewer for
                   $k = 3$ \\
    \bottomrule
  \end{tabular}
  \vskip 0.6em
  \begin{itemize}
    \item $H$ and $W$ are the size of the output map, $k$ is the kernel
          size, and $C$ is a number of channels.
  \end{itemize}
  \keypoint{A model has four controls for its cost. An efficient design
    sets all four for the device.}
\end{frame}

\begin{frame}{The MobileNet family}
  \small
  \renewcommand{\arraystretch}{1.3}
  \begin{tabular}{@{}L{0.20\textwidth}L{0.10\textwidth}L{0.60\textwidth}@{}}
    \toprule
    \textbf{Model} & \textbf{Year} & \textbf{The new idea} \\
    \midrule
    MobileNetV1 & 2017 & Depthwise separable convolutions in each layer.
        Two controls: the width multiplier and the resolution. \\
    MobileNetV2 & 2018 & The inverted residual block: narrow, wide,
        narrow, with a skip connection between the narrow tensors
        (Day 5). \\
    MobileNetV3 & 2019 & The blocks and the widths come from a search with
        the measured latency of a phone. \\
    \bottomrule
  \end{tabular}
  \vskip 0.6em
  \begin{itemize}
    \item The family is the standard start for vision on a small device.
          You select two numbers: the width multiplier and the image size.
  \end{itemize}
  \keypoint{Take a known efficient family, and set its controls for your
    budgets. This is faster than a new design, and it is usually better.}
  \source{Howard et al., 2017; Sandler et al., 2018; Howard et al., 2019.}
\end{frame}

\begin{frame}{EfficientNet: scale three dimensions together}
  \begin{columns}[c]
    \begin{column}{0.46\textwidth}
      \[
        d = \alpha^{\phi} \qquad w = \beta^{\phi} \qquad r = \gamma^{\phi}
      \]
      \small
      \begin{tabular}{@{}rL{0.74\textwidth}@{}}
        $d$ & Factor for the depth \\
        $w$ & Factor for the width \\
        $r$ & Factor for the image size \\
        $\phi$ & The size step that you select \\
      \end{tabular}
    \end{column}
    \begin{column}{0.52\textwidth}
      \begin{itemize}
        \item The MACs grow with $d \cdot w^2 \cdot r^2$.
        \item A larger image needs more layers and more channels to use
              its detail. So the three dimensions must grow together.
        \item The paper uses $\alpha = 1.2$, $\beta = 1.1$, and
              $\gamma = 1.15$. Each step of $\phi$ gives about two times
              the MACs.
      \end{itemize}
    \end{column}
  \end{columns}
  \vskip 0.5em
  \small
  \renewcommand{\arraystretch}{1.2}
  \begin{tabular}{@{}L{0.24\textwidth}rrr@{}}
    \toprule
    \textbf{Model} & \textbf{Parameters} & \textbf{Operations}
    & \textbf{ImageNet accuracy} \\
    \midrule
    ResNet-50        & 26 million  & 4.1 billion  & 76.0 percent \\
    EfficientNet-B0  & 5.3 million & 0.39 billion & 77.1 percent \\
    \bottomrule
  \end{tabular}
  \source{Tan and Le, ``EfficientNet: Rethinking Model Scaling for
    Convolutional Neural Networks'', 2019.}
\end{frame}

\begin{frame}{Fewer operations are not always less time}
  \small
  \renewcommand{\arraystretch}{1.25}
  \begin{tabular}{@{}L{0.36\textwidth}rr@{}}
    \toprule
    \textbf{Against the baseline} & \textbf{MACs} & \textbf{Latency} \\
    \midrule
    Pruned to 8, 16, 32 filters & 3.8 times fewer & 2.7 times shorter \\
    Separable layers            & 5.4 times fewer & 2.6 times shorter \\
    Baseline in \code{int8}     & the same        & 1.6 times longer \\
    \bottomrule
  \end{tabular}
  \vskip 0.5em
  \begin{itemize}
    \item The time has parts that the MACs do not count: the memory
          access, the start of each kernel, and the copy of the input.
    \item On this laptop processor, the \code{int8} kernels are slower than
          the float kernels. On a microcontroller with no float unit, the
          result is the opposite.
    \item The book reports cases where the operation count and the latency
          differ by a factor of 3 to 5.
  \end{itemize}
  \keypoint{The operation count is an estimate for the design. The latency
    on the device is the result.}
  \source{The example model with LiteRT on a laptop CPU, one thread. Factor
    of 3 to 5: \emph{Machine Learning Systems}, Volume I, chapter 10.}
\end{frame}

\begin{frame}{Neural architecture search}
  \begingroup\centering
  \begin{tikzpicture}[font=\footnotesize, >={Stealth[length=5pt]},
      b/.style={draw=coolgray, rounded corners=3pt, fill=boxgray,
                minimum width=2.3cm, minimum height=1.0cm, align=center},
      lab/.style={font=\scriptsize, text=coolgray, align=center}]
    \node[b] (a) at (0,0)   {Search\\space};
    \node[b, draw=darkcyan] (b) at (3.4,0) {Search\\strategy};
    \node[b, draw=cardinalred] (c) at (6.8,0) {Evaluation};
    \draw[->] (a) -- (b);
    \draw[->] (b) -- node[above, lab] {candidate} (c);
    \draw[->] (c.south) -- ++(0,-0.55) -| node[pos=0.25, below, lab]
      {accuracy and cost of the candidate} (b.south);
    \node[lab] at (0,-0.95) {which layers, widths,\\and kernel sizes};
  \end{tikzpicture}\par\endgroup
  \vskip 0.5em
  \begin{itemize}
    \item A program designs the architecture. It tests many candidates and
          keeps the best.
    \item \textbf{Search space:} the building blocks and their options, for
          example a block with a kernel of 3, 5, or 7 and an expansion of
          3 or 6.
    \item \textbf{Search strategy:} how the program selects the next
          candidate.
    \item \textbf{Evaluation:} the accuracy of the candidate, and its cost.
  \end{itemize}
  \keypoint{The search needs a trained model for each candidate. The
    evaluation is the expensive step.}
  \source{\emph{Machine Learning Systems}, Volume I, chapter 10.}
\end{frame}

\begin{frame}{The cost of the search}
  \small
  \renewcommand{\arraystretch}{1.3}
  \begin{tabular}{@{}L{0.26\textwidth}L{0.38\textwidth}L{0.26\textwidth}@{}}
    \toprule
    \textbf{Strategy} & \textbf{How it works} & \textbf{Reported cost} \\
    \midrule
    Reinforcement learning & A controller proposes a model and gets its
        accuracy as the reward & 22\,400 GPU-days for 12\,800 models \\
    Evolution & A population of models. Change the best models and select
        again. & 3150 GPU-days \\
    Gradient (DARTS) & All options are in one large network with weights
        for each option & 1 to 4 GPU-days \\
    \bottomrule
  \end{tabular}
  \vskip 0.6em
  \begin{itemize}
    \item The first searches trained each candidate from the start. Weight
          sharing trains one large network one time, and each candidate
          uses a part of it. The cost falls by a factor of about 1000.
  \end{itemize}
  \keypoint{A search is for a product with millions of devices. All others
    use the models that the searches found.}
  \source{\emph{Machine Learning Systems}, Volume I, chapter 10.}
\end{frame}

\begin{frame}{Hardware-aware search}
  \[
    \text{score} = \text{accuracy} \times
    \left(\frac{\text{latency}}{\text{target latency}}\right)^{\beta}
    \qquad \beta < 0
  \]
  \begin{itemize}
    \item The score has two parts: the accuracy and the latency on the
          target device.
    \item A candidate that is slower than the target loses score. A
          candidate that is faster wins a little.
    \item The latency is a measurement on the device, or a table with the
          measured time of each block. The search then prefers layers that
          are fast on this device.
  \end{itemize}
  \vskip 0.3em
  \begin{block}{A reported result}
    A search with the latency of a phone in the score found models that are
    1.8 times as fast as MobileNetV2, with a higher accuracy.
  \end{block}
  \source{Score and result of MnasNet (Tan et al., 2019), from \emph{Machine
    Learning Systems}, Volume I, chapter 10.}
\end{frame}

\begin{frame}{MCUNet: a search for microcontrollers}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{The model: TinyNAS}
        \begin{itemize}
          \item First it fits the search space to the device: it selects
                the image size and the width for the RAM and the flash.
          \item Then it searches a model in this space.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{The runtime: TinyEngine}
        \begin{itemize}
          \item It generates code for the layers of this one model only.
          \item It plans the memory for the complete model, not for each
                layer alone.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
  \vskip 0.5em
  \begin{itemize}
    \item The limit of the search is the peak activation memory of Day 5,
          not only the size of the weights.
    \item The model and the runtime are designed together. A better memory
          plan of the runtime permits a larger model.
    \item The result: more than 70 percent ImageNet accuracy on a
          commercial microcontroller.
  \end{itemize}
  \source{Lin et al., ``MCUNet: Tiny Deep Learning on IoT Devices'', 2020.}
\end{frame}

\begin{frame}{The order: prune, distill, quantize}
  \begingroup\centering
  \begin{tikzpicture}[font=\footnotesize, >={Stealth[length=5pt]},
      b/.style={draw=coolgray, rounded corners=3pt, fill=boxgray,
                minimum width=1.75cm, minimum height=1.0cm, align=center},
      lab/.style={font=\scriptsize, text=coolgray, align=center}]
    \node[b] (a) at (0,0)    {Trained\\model};
    \node[b, draw=cardinalred] (b) at (2.15,0) {Change the\\structure};
    \node[b] (c) at (4.3,0)  {Fine-tune\\or train};
    \node[b, draw=darkcyan] (d) at (6.45,0) {Quantize};
    \node[b] (e) at (8.6,0)  {Measure on\\the device};
    \foreach \a/\b in {a/b, b/c, c/d, d/e} { \draw[->] (\a) -- (\b); }
    \node[lab] at (2.15,-0.95) {prune, distill,\\or factorize};
    \node[lab] at (4.3,-0.95)  {in \code{float32}};
    \node[lab] at (6.45,-0.95) {the last step};
  \end{tikzpicture}\par\endgroup
  \vskip 0.4em
  \begin{itemize}
    \item The structural methods need training with gradients. Training
          works best in \code{float32}.
    \item Quantization needs the final weights and the final activations:
          the calibration measures their ranges. A change of the structure
          after it makes the ranges wrong.
    \item If the \code{int8} model loses too much, the last training step
          is quantization-aware training (Day 4).
    \item After each step, measure the accuracy. Then you know which step
          caused a loss.
  \end{itemize}
  \keypoint{First decide what the model calculates. Then decide how it
    stores the numbers.}
\end{frame}

\begin{frame}{A method for the selection}
  \begin{enumerate}
    \item \textbf{Write the budgets} of the device: flash, RAM, latency,
          and the smallest accuracy that the product accepts.
    \item \textbf{Measure the baseline} in \code{float32} on the device, or
          estimate it: file size, peak of the activations, MACs.
    \item \textbf{Quantize first.} Post-training quantization needs
          seconds, and it helps two or three budgets.
    \item \textbf{Find the budget that still fails.} Select the technique
          for this budget from the table.
    \item \textbf{Apply the cheapest technique first:} a smaller image or a
          smaller width, then structured pruning, then distillation, then
          a new architecture.
    \item \textbf{Measure all budgets again} after each step. A step can
          repair one budget and break a different one.
  \end{enumerate}
  \vskip 0.3em
  \keypoint{The selection is a loop of measurements, not one decision. Stop
    at the first model that meets all budgets.}
\end{frame}

\exerciseframe{select techniques for three cases}{%
  \small
  One vision model, three products. The model in \code{float32}:
  \vskip 0.1em
  {\footnotesize
  \renewcommand{\arraystretch}{1.05}
  \begin{tabular}{@{}L{0.46\textwidth}r@{}}
    \toprule
    Weights & 1.2 million parameters: 4.6 MB \\
    Peak of the activations & 600\,000 values: 2.3 MB \\
    Latency on the target processor & 400 ms \\
    Image size & $160 \times 160$ \\
    \bottomrule
  \end{tabular}}
  \vskip 0.2em
  \begin{enumerate}
    \item \textbf{Flash limit.} The device has 1.5 MB of flash for the
          model. The RAM and the latency are no problem.
    \item \textbf{RAM limit.} The device has 256 KB of RAM for the
          activations. The flash is no problem.
    \item \textbf{Latency limit.} The product needs 150 ms. The device has
          a float unit and no optimized \code{int8} kernels.
  \end{enumerate}
  For each case: name the first technique, calculate its effect, and say
  if a second technique is necessary. Use the factors of this day: 4 for
  \code{int8}, and the square of the side for the image size.
  \vskip 0.2em
  Work with your neighbour for 10 minutes.
}

\answerframe{select techniques for three cases}{%
  \small
  \begin{enumerate}
    \item \textbf{Flash.} Quantize to \code{int8}: $4.6 / 4 = 1.15$ MB.
          This is below 1.5 MB. No second technique is necessary. Check the
          accuracy.
    \item \textbf{RAM.} Quantize to \code{int8}: the peak becomes
          $600\,000$ bytes, or 586 KB. This is still above 256 KB. Then
          use a smaller image. With $96 \times 96$, the factor is
          $(96 / 160)^2 = 0.36$: 211 KB. This fits. Structured pruning of
          the first layers is the alternative. Unstructured pruning and
          separable layers do not help.
    \item \textbf{Latency.} The target is $150 / 400 = 0.375$ of the time.
          Quantization does not help on this device. Reduce the MACs:
          the image size $96 \times 96$ gives a factor of 0.36, or
          structured pruning to 60 percent of the channels gives
          $0.6^2 = 0.36$. The time falls less than the MACs, so plan a
          margin, and measure. If the accuracy is then too low, distill
          into the smaller model with the large model as the teacher.
  \end{enumerate}
  \vskip 0.1em
  One technique repairs cases 2 and 3 together: the smaller image. It is
  also the cheapest. Measure the accuracy first.
}
