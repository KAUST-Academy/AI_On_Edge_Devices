\begin{frame}[fragile]{An inference is a long list of multiply-accumulates}
\begin{lstlisting}[style=cpp]
// One convolution of 3 x 3. Each line "acc +=" is one multiply-accumulate (MAC).
for (int oc = 0; oc < C_out; oc++)
  for (int y = 0; y < H; y++)
    for (int x = 0; x < W; x++) {
      float acc = bias[oc];
      for (int ic = 0; ic < C_in; ic++)
        for (int ky = 0; ky < 3; ky++)
          for (int kx = 0; kx < 3; kx++)
            acc += in[ic][y + ky][x + kx] * w[oc][ic][ky][kx];
      out[oc][y][x] = acc;
    }
\end{lstlisting}
  \small
  \begin{itemize}
    \item MobileNetV2 for an image of $224 \times 224$ needs 300.8 million
          MACs.
    \item With one MAC in each clock cycle, the Raspberry Pi 5 needs
          $300.8 \times 10^6 / (2.4 \times 10^9)$ seconds: 125 ms.
    \item A good runtime is much faster than this. You measure the real
          value in the lab. This part explains where the gain comes from.
  \end{itemize}
  \source{A calculation with one MAC in each clock cycle of 2.4 GHz. Number
    of MACs: the notebook of Day 1.}
\end{frame}

  \end{tikzpicture}\par\endgroup

    \item The three ways do the same MACs in less time. They do not change
          the model.
    \item The methods of Days 4 and 6 are the fourth way: fewer MACs, or

% ------------------------------------------------------------------ threads
\begin{frame}{Threads, and the law of Amdahl}
  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \[
        \text{speedup} = \frac{1}{(1 - p) + \dfrac{p}{G}}
      \]
      \small
      \begin{tabular}{@{}rL{0.78\textwidth}@{}}
        $p$ & The part of the time that the faster path does \\
        $G$ & The gain on this part: 4 for 4 cores \\
      \end{tabular}
    \end{column}
    \begin{column}{0.48\textwidth}
      \small
      \renewcommand{\arraystretch}{1.15}
      \begin{tabular}{@{}rrr@{}}
        \toprule
        $p$ & \textbf{4 cores} & \textbf{No limit on} $G$ \\
        \midrule
        0.50 & 1.6 & 2 \\
        0.80 & 2.5 & 5 \\
        0.90 & 3.1 & 10 \\
        0.95 & 3.5 & 20 \\
        \bottomrule
      \end{tabular}
    \end{column}
  \end{columns}
  \vskip 0.6em
  \begin{itemize}
    \item A runtime splits the work of a layer between the cores. The
          Raspberry Pi 5 has 4 cores.
    \item Some work stays outside: read the camera, change the size of the
          image, the Python code, the post-processing.
    \item The same law applies to each accelerator. A unit that makes 90
          percent of the time 100 times faster gives a speedup of 9.2.
  \end{itemize}
  \keypoint{Measure the complete pipeline, from the camera to the result.
    The part that you do not accelerate sets the limit.}
\end{frame}

\begin{frame}{The GPU: many simple cores}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{A CPU core}
        \begin{itemize}
          \item It is made for code with branches.
          \item A large part of its area is control logic and cache.
          \item A small part does arithmetic.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{A GPU}
        \begin{itemize}
          \item It is made for one simple step on very many values.
          \item Most of its area does arithmetic.
          \item One thread is slow. Thousands together are fast.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
  \vskip 0.5em
  \begin{itemize}
    \item Each call of a GPU has a price: the program sends the data and
          starts the kernel. For a small model and one image, the CPU can
          be faster.
    \item The Raspberry Pi 5 has a GPU for graphics. The labs run the
          models on the CPU.
  \end{itemize}
  \keypoint{A GPU pays for large tensors and large batches. An edge device
    often has one image and a small model.}
\end{frame}

  \end{itemize}
  \keypoint{Before you buy an accelerator, check that it supports each
    operator of your model.}

% ----------------------------------------------------------- the two regimes
\begin{frame}{Compute-bound or memory-bound}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Compute-bound}
        \begin{itemize}
          \item The arithmetic units are busy all the time.
          \item The data arrives fast enough.
          \item A faster core, wider SIMD, or more cores help.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Memory-bound}
        \begin{itemize}
          \item The arithmetic units wait for data.
          \item The memory bus is busy all the time.
          \item Only fewer bytes, or a faster memory, help.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
  \vskip 0.6em
  \begin{itemize}
    \item One model has layers of the two types. A large convolution does
          many MACs with each byte. A ReLU does one comparison and moves
          8 bytes.
    \item The wrong optimization gives no gain. More cores do not help a
          workload that waits for the memory.
  \end{itemize}
  \keypoint{First find the limit of the workload. Then select the
    optimization.}
\end{frame}

\begin{frame}{Arithmetic intensity}
  \[
    I = \frac{\text{operations}}{\text{bytes that move}}
    \qquad \text{unit: FLOP for each byte}
  \]
  \footnotesize
  \renewcommand{\arraystretch}{1.1}
  \begin{tabular}{@{}L{0.62\textwidth}r@{}}
    \toprule
    \textbf{Operator in \texttt{float32}} & $I$ \\
    \midrule
    Add two feature maps                                       & 0.083 \\
    ReLU                                                       & 0.125 \\
    Fully connected layer, one input vector                    & 0.5 \\
    Depthwise convolution $3 \times 3$                         & 2.25 \\
    Convolution $3 \times 3$, 64 to 64 channels, $56 \times 56$ & 132 \\
    \bottomrule
  \end{tabular}
  \vskip 0.3em
  \small
  \begin{itemize}
    \item One MAC is 2 operations. The bytes are the input, the weights,
          and the output: each tensor one time.
    \item A fully connected layer uses each weight one time. A convolution
          uses each weight at each position of the image.
    \item In \code{int8}, the same operator moves one quarter of the bytes.
          Its intensity is 4 times as high.
  \end{itemize}
  \source{A calculation. Method: \emph{Machine Learning Systems}, Volume I,
    chapter 11.}
\end{frame}

        \fill[cardinalred!8] (2.3,0) rectangle (5.4,2.7);
        \draw[darkcyan, line width=1.2pt] (0,0.4) -- (2.3,2.7);
        \draw[cardinalred, line width=1.2pt] (2.3,2.7) -- (5.4,2.7);
        \draw[coolgray, dashed] (2.3,0) -- (2.3,2.7);
        \node[darkcyan!80!black, rotate=45, anchor=south] at (1.05,1.5)
          {$\text{BW} \times I$};
        \node[cardinalred, anchor=south] at (3.9,2.7) {$R_{\text{peak}}$};
        \node[darkcyan!80!black] at (1.25,0.5) {memory-bound};
        \node[cardinalred] at (3.9,1.2) {compute-bound};
        \node[coolgray, anchor=west] at (2.35,0.2) {ridge point};
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.44\textwidth}
      \[
        R = \min\left(R_{\text{peak}},\; \text{BW} \times I\right)
      \]
      \small
      \begin{tabular}{@{}rL{0.76\textwidth}@{}}
        $R$ & The highest rate that the workload can reach \\
        $R_{\text{peak}}$ & The peak rate of the arithmetic units \\
        BW & The memory bandwidth in bytes for each second \\
      \end{tabular}
    \end{column}
  \end{columns}
  \vskip 0.4em
  \small
  \begin{itemize}
    \item The two axes have a logarithmic scale. The roof has two parts: a
          slope and a flat part.
    \item The \textbf{ridge point} is $R_{\text{peak}} / \text{BW}$. A
          workload with a lower intensity is memory-bound.
    \item The hardware gives the roof. The workload gives the position
          below the roof.
  \end{itemize}
  \source{Williams, Waterman, and Patterson, ``Roofline: An Insightful
    Visual Performance Model for Multicore Architectures'', 2009.}
\end{frame}

\begin{frame}{Read the roofline with care}
  \small
  \renewcommand{\arraystretch}{1.3}
  \begin{tabular}{@{}L{0.30\textwidth}L{0.62\textwidth}@{}}
    \toprule
    \textbf{Fact} & \textbf{Example of the measurement} \\
    \midrule
    The roof is a limit. It is not a prediction.
      & The matrix product with three simple loops reaches 0.91 GFLOP/s.
        The library reaches 82.8 with the same arithmetic: a factor of
        about 90. \\
    The intensity depends on the code.
      & The library cuts the matrices into blocks that fit in a cache. It
        uses each byte many times before the byte leaves the cache. \\
    The bandwidth depends on the place of the data.
      & The loop $y \mathrel{+}= a\,x$ has an intensity of 0.167. With arrays in the
        first cache it reaches 22.2 GFLOP/s. The roof for data in the
        RAM is 2.4. \\
    \bottomrule
  \end{tabular}
  \vskip 0.6em
  \begin{itemize}
    \item A very small model has one more limit: the fixed cost of each
          operator in the runtime.
  \end{itemize}
  \keypoint{The roof tells you which resource limits a good kernel. The
    quality of the kernel decides how near to the roof you get.}
  \source{The experiment of this course: an x86 processor with AVX2, one
    thread.}
\end{frame}

\begin{frame}{The regime selects the optimization}
  \small
  \renewcommand{\arraystretch}{1.3}
  \renewcommand{\arraystretch}{1.2}
  \begin{tabular}{@{}L{0.15\textwidth}L{0.39\textwidth}L{0.38\textwidth}@{}}
    \toprule
    & \textbf{Memory-bound} & \textbf{Compute-bound} \\
    \midrule
    Typical operators
      & ReLU, add, a fully connected layer, a depthwise convolution
      & A large convolution, a matrix product with a batch \\
    What helps
      & Fewer bytes: \code{int8}, smaller feature maps. Fused operators
        (Part 3).
      & SIMD kernels, threads, an accelerator. Fewer MACs (Day 6). \\
    No help
      & More cores, a higher peak rate
      & A faster memory \\
    \bottomrule
  \end{tabular}
  \vskip 0.4em
  \begin{itemize}
    \item \code{int8} helps in the two regimes: one quarter of the bytes,
          and 4 times as many values in one register.
    \item This is the limit of the hardware. The kernels decide what you
          get: on the work computer of Day 6, the \code{int8} models were
          1.6 times slower.
    \item A language model that makes one token reads all its weights for
          this token. It is memory-bound (Day 10).
  \end{itemize}
\end{frame}

\exerciseframe{place three workloads on a Roofline plot}{%
  \small
  One core of an example device has a peak rate of 40 GFLOP/s and a memory
  bandwidth of 8 GB/s. Three layers in \code{float32} run on it. The table
  gives the operations and the bytes in millions.
  \vskip 0.2em
  {\footnotesize
  \renewcommand{\arraystretch}{1.1}
  \setlength{\tabcolsep}{4pt}
  \begin{tabular}{@{}lL{0.66\textwidth}rr@{}}
    \toprule
    & \textbf{Layer} & \textbf{Operations} & \textbf{Bytes} \\
    \midrule
    A & Convolution $3 \times 3$, 64 to 64 channels, $56 \times 56$
      & 231.2 & 1.75 \\
    B & Depthwise convolution $3 \times 3$, 32 channels, $112 \times 112$
      & 7.23 & 3.21 \\
    C & Fully connected, 1280 to 1000 & 2.56 & 5.13 \\
    \bottomrule
  \end{tabular}}
  \begin{enumerate}
    \item Calculate the ridge point of the device. Draw the roof: the axes
          are the intensity from 0.1 to 1000, and the rate from 1 to 100
          GFLOP/s.
    \item Calculate the arithmetic intensity of each layer. Place the three
          layers on the plot, at the roof.
    \item Which layers are memory-bound? Calculate the shortest time of
          each layer.
    \item The model changes to \code{int8}: each layer moves one quarter of
          the bytes. Which layer changes its regime?
  \end{enumerate}
}
