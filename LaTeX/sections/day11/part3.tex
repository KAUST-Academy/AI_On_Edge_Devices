% ------------------------------------------------------------------ locality
\begin{frame}{The data locality invariant}
  \small
  The textbook compares two complete paths with the deadline of the
  service:
  \[
    T_{\text{remote}} = \frac{D}{\text{BW}} + L_{\text{network}} +
      \frac{O}{R_{\text{remote}} \cdot \eta}
    \qquad\text{remote is possible only if } T_{\text{remote}} \leq
      T_{\text{deadline}}
  \]
  \vskip 0.2em
  \footnotesize
  \renewcommand{\arraystretch}{1.12}
  \begin{tabular}{@{}lL{0.70\textwidth}@{}}
    \toprule
    \textbf{Symbol} & \textbf{Meaning} \\
    \midrule
    $D$                  & The data of one request: a frame, or a part of a
                           stream \\
    $\text{BW}$          & The bandwidth to the remote node \\
    $L_{\text{network}}$ & The round trip of the network (Part 1) \\
    $O$, $R \cdot \eta$  & The operations of the model, and the real rate of
                           the remote processor \\
    \bottomrule
  \end{tabular}
  \vskip 0.4em
  \small
  \begin{itemize}
    \item Compute $T_{\text{local}}$ in the same way: the model time on the
          camera node. Then compare the paths that meet the deadline.
    \item The result is small: a class or a few boxes. Its transfer time
          is almost zero.
    \item A faster remote processor does not help when the transfer is the
          largest part.
  \end{itemize}
  \source{\emph{Machine Learning Systems}, Volume I, chapter ``ML
    Systems'', section ``The data locality invariant''.}
\end{frame}

% ------------------------------------------------------------------ bandwidth
\begin{frame}{Bandwidth: send the video or the results}
  \small
  12 cameras on one Wi-Fi channel:
  \vskip 0.3em
  \footnotesize
  \renewcommand{\arraystretch}{1.15}
  \begin{tabular}{@{}L{0.42\textwidth}rr@{}}
    \toprule
    \textbf{What the camera node sends} & \textbf{One camera} &
    \textbf{12 cameras} \\
    \midrule
    H.264, 1280 $\times$ 720, keyframe each 1 s (Part 2) & 2.62 Mbit/s &
      31.4 Mbit/s \\
    MJPEG, 1280 $\times$ 720, quality 5 (Part 2) & 11.09 Mbit/s & 133.1
      Mbit/s \\
    Results: 10 frames each second, 300 bytes of JSON each & 0.024 Mbit/s &
      0.29 Mbit/s \\
    \bottomrule
  \end{tabular}
  \vskip 0.5em
  \small
  \begin{itemize}
    \item The results need about 100 times fewer bits than the H.264
          stream.
    \item 12 MJPEG streams do not fit one Wi-Fi channel. 12 H.264 streams
          use most of it, and the channel then has no room for other
          traffic.
    \item With the results only, the channel stays almost free, and a
          device with a weak signal does not slow the others.
  \end{itemize}
  \keypoint{Each camera that sends video uses the shared channel. A camera
    that sends results uses almost nothing.}
  \source{Bit rates: Part 2 (moving camera).}
\end{frame}

\begin{frame}{Where the age of a frame comes from}
  \begingroup\centering
  \begin{tikzpicture}[font=\scriptsize, x=1cm]
    \foreach \x/\w/\t/\c in {0/1.1/Capture/boxgray, 1.1/1.2/Encode/cardinalred!15,
        2.3/1.2/Send/darkcyan!20, 3.5/1.4/Buffers/darkcyan!35,
        4.9/1.4/Decode/cardinalred!15, 6.3/1.2/Queue/darkcyan!35,
        7.5/1.6/Model/cardinalred!30, 9.1/1.0/Show/boxgray} {
      \draw[coolgray, fill=\c] (\x,0) rectangle ++(\w,0.7);
      \node at (\x+\w/2,0.35) {\t};
    }
    \draw[<->, coolgray] (0,-0.25) -- node[below] {age of the frame when the
      result appears} (10.1,-0.25);
  \end{tikzpicture}
  \par\endgroup
  \vskip 0.4em
  \small
  \begin{itemize}
    \item \textbf{Capture}: the exposure and the transfer from the sensor.
    \item \textbf{Encode} and \textbf{decode}: B-frames and decoder threads
          hold frames.
    \item \textbf{Buffers}: the server queue, the TCP buffers, and the
          jitter buffer of the reader protect against changes of the
          network delay. Each one holds frames.
    \item \textbf{Queue}: frames that wait for a slow model.
    \item A buffer smooths the stream, and it makes each frame older. A live
          system keeps the buffers small.
  \end{itemize}
  \keypoint{The latency of the model is one box of eight. Measure the age of
    the frame at the moment of the result.}
\end{frame}

\begin{frame}{A slow model behind a fast stream}
  \begingroup\centering
  \begin{tikzpicture}[font=\scriptsize, >=stealth]
    \foreach \i in {0,...,8} {
      \draw[coolgray, fill=darkcyan!20] (\i*0.45,0) rectangle ++(0.38,0.6);
    }
    \node[below] at (1.95,-0.05) {queue};
    \draw[->, darkcyan] (-2.1,0.3) -- node[above] {30 frames/s in}
      (-0.1,0.3);
    \node[draw=coolgray, rounded corners=2pt, fill=cardinalred!15,
          minimum width=1.6cm, minimum height=0.8cm] (m) at (5.4,0.3)
      {Model};
    \draw[->] (4.0,0.3) -- (m);
    \draw[->, cardinalred] (m.east) -- node[above] {10 frames/s out}
      ++(2.1,0);
  \end{tikzpicture}
  \par\endgroup
  \vskip 0.5em
  \small
  \begin{itemize}
    \item A model of 100 ms takes 10 frames each second. The stream brings
          30.
    \item A reader that takes each frame in order adds 20 frames to its
          queue each second. Each frame waits longer than the frame before.
    \item The age grows by $1 - 10/30 = 0.67$ s each second, with no
          limit until a buffer is full.
    \item A full buffer then drops data at a place that the program does
          not control: a packet in the server, or a frame in the reader.
  \end{itemize}
  \keypoint{When the model is slower than the stream, the program must drop
    frames. Select which frames it drops.}
\end{frame}

\begin{frame}{Drop frames on purpose}
  \footnotesize
  \renewcommand{\arraystretch}{1.18}
  \begingroup\centering
  \begin{tabular}{@{}L{0.32\textwidth}L{0.60\textwidth}@{}}
    \toprule
    \textbf{Method} & \textbf{Effect} \\
    \midrule
    Keep the newest frame & A thread reads all frames; the model takes the
      last one. The lab reader does this. \\
    A lower frame rate at the camera & 10 frames each second for a model of
      100 ms: less encoder work and fewer bits \\
    A second, smaller stream & MediaMTX can send a second stream of the same
      camera with its own size and rate \\
    Skip a fixed number of frames & Simple, but the queue still grows when
      the model time changes \\
    \bottomrule
  \end{tabular}\par\endgroup
  \vskip 0.5em
  \small
  \begin{itemize}
    \item A result for an old frame is a wrong result for a moving object.
          A person who walks at 1.4 m/s moves 9.4 m in 6.7 s.
    \item A dropped frame is a choice of the program. A full buffer drops
          data with no choice.
    \item Report two numbers: the frames each second that the model takes,
          and the age of those frames.
  \end{itemize}
  \source{MediaMTX documentation, ``Raspberry Pi Cameras''.}
\end{frame}

% ------------------------------------------------------------------ privacy
\begin{frame}{Video shows people}
  \small
  \begin{itemize}
    \item A camera records faces, number plates, clothes, places, and
          behaviour. Video of people is personal data in many laws, for
          example the GDPR of the European Union and the Personal Data
          Protection Law of Saudi Arabia.
    \item A stream can show more than its task needs: a person counter
          does not need the face of the person.
    \item Each copy is a risk: the stream on the network, a recording on a
          disk, a frame in a log, a test image on a laptop.
  \end{itemize}
  \vskip 0.3em
  Questions before you install a camera:
  \begin{enumerate}
    \item Which result does the task need? A count, a class, a box, or an
          image?
    \item Where does each image go, and who can see it?
    \item How long does each copy stay? Who deletes it?
    \item Do the people know about the camera?
  \end{enumerate}
  \keypoint{Collect the least data that the task needs. An edge device can
    send the result and keep the image.}
\end{frame}

\begin{frame}{Protect a stream}
  \footnotesize
  \renewcommand{\arraystretch}{1.18}
  \begingroup\centering
  \begin{tabular}{@{}L{0.26\textwidth}L{0.32\textwidth}L{0.34\textwidth}@{}}
    \toprule
    \textbf{Risk} & \textbf{The lab setup} & \textbf{A product} \\
    \midrule
    Anybody on the network watches & No password: each device on
      \code{edgeai-lab} can open the stream & User names and passwords in
      MediaMTX \\
    Somebody reads the packets & RTSP and RTP with no encryption & RTSPS,
      and SRTP for the packets, with a TLS certificate \\
    Images leave the site & A closed lab network & Results only; images on
      request \\
    Old recordings & None & A fixed time, then delete \\
    \bottomrule
  \end{tabular}\par\endgroup
  \vskip 0.5em
  \small
  \begin{itemize}
    \item The lab network is closed, and the cameras show the lab. This is
          why the lab needs no password.
    \item Encryption costs CPU time on the camera node. Measure it, like
          the encoder.
    \item The MJPEG stream of the XIAO has no password and no encryption.
          Use it only on a closed network.
  \end{itemize}
  \source{MediaMTX documentation, ``RTSP-specific features'' (section
    ``Encryption'') and ``Authentication''.}
\end{frame}

    \draw[->] (c) -- (m); \draw[->] (m) -- (r); \draw[->] (r) -- (d);
    \node[st, fill=white, text width=2.4cm] (b) at (2.7,-1.45) {Image: blur

    \draw[->, coolgray] (m) -- (b);
  \end{tikzpicture}
  \par\endgroup
  \vskip 0.4em
  \small
  \begin{itemize}
    \item \textbf{Send results, not images.} The detector runs where the
          image is made.
    \item \textbf{Change the image before it leaves.} Blur faces and number
          plates on the camera node, then stream.
    \item \textbf{Store events, not video.} Keep a short clip for an event,
          and delete it after a fixed time.
    \item \textbf{Limit the access.} Passwords for the stream, a separate
          network for the cameras, a log of each access.
  \end{itemize}
  \keypoint{The place of the model is also a privacy decision. An image that
    never leaves the device cannot leak from the network.}
\end{frame}

% -------------------------------------------------------------------- exercise
\exerciseframe{select the inference location}{%
  \small
  For each system, select A (model on the camera node), B (model on a
  central node), or C (split). Give two reasons: one number and one other
  reason.
  \begin{enumerate}
    \item \textbf{Parking site.} 8 cameras count the free places each
          minute. The site has an internet link of 5 Mbit/s. No image of a
          car may leave the site.
    \item \textbf{Doorbell.} A camera on a battery. It must show a video on
          the phone of the owner when a person rings. Most of the day,
          nothing happens.
    \item \textbf{Factory line.} One camera checks each part for a defect.
          The belt must stop within 100 ms. A strong computer stands in the
          same room, with a cable to the camera.
  \end{enumerate}
  Use the bit rates of Part 2 (2.62 Mbit/s for H.264) and the frame ages of
  this part.
}

\answerframe{select the inference location}{%
  \small
  \begin{enumerate}
    \item \textbf{A, or B with a node on the site.} 8 H.264 streams need
          $8 \times 2.62 = 21$ Mbit/s: 4 times the link. A count each
          minute is a few bytes. The images must stay on the site.
    \item \textbf{C.} A small model or a motion test on the camera wakes
          the radio and the stream only for a person. The battery cannot
          send video all day. The video goes to the phone, with
          encryption.
    \item \textbf{B is possible, A is safer.} A stream with 1 decoder
          thread had an age of 40 ms on one computer; the cable, the model,
          and the stop signal must fit into the other 60 ms. With 4 decoder
          threads (139 ms), B fails. A model on the camera node removes the
          video path.
  \end{enumerate}
  \keypoint{Write the time budget and the bit rate first. Then select the
    place, and add the privacy and the power.}
}
