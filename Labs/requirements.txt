# Python packages for the labs of "AI on Edge Devices".

# Base. Every lab uses these.
numpy
matplotlib
pandas
jupyterlab

# Training and profiling of models (Days 1, 3, 4, and 6).

#   .venv/bin/pip install torch torchvision --index-url https://download.pytorch.org/whl/cpu
torch
torchvision

tensorflow-cpu
ai-edge-litert
onnx
onnxruntime
